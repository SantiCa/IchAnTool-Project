"""Batch processing: many images, and cores made of several sections, from a manifest table.

The manifest is a CSV with one row per section image::

    image,core,top,bottom,units,name,exclude
    U1385_1H_1.png,U1385I,0,150,cm,1H-1,U1385_1H_1_excl.png

``image``, ``top`` and ``bottom`` are required; ``core`` (default: one core named after the manifest), ``units``
(default cm), ``name`` (default: file stem) and ``exclude`` (an exclusion image, white = excluded) are optional.
Relative paths are read from the manifest's folder. Write a starting manifest for a folder with
:func:`write_manifest_template`.
"""

from __future__ import annotations

from collections.abc import Callable
from pathlib import Path

import numpy as np
import pandas as pd

from .export import spreadsheet_text, write_csv_table
from .io import SUPPORTED, load_image, load_mask
from .pipeline import UNIT_TO_CM, CoreSection, analyze_mask

REQUIRED = ("image", "top", "bottom")
_MANIFEST_ESCAPE = "__sedcoretool_csv_escape__"
_ESCAPE_VERSION = "apostrophe-v1"
_MANIFEST_TYPES = {**dict.fromkeys(("image", "core", "name", "exclude", "units", _MANIFEST_ESCAPE), "string"),
                   "top": "float64", "bottom": "float64"}


def load_exclusion(path: str | Path) -> np.ndarray:
    """Exclusion image: pixels with grey > 127 (white) are excluded, black pixels are valid."""
    img = load_image(path, "L")
    return img > 127


def _read_matching_exclusion(path: str | Path, image_path: str | Path, image_shape: tuple[int, ...]) -> np.ndarray:
    exclusion = load_exclusion(path)
    if exclusion.shape != image_shape:
        raise ValueError(f"{Path(path).name}: size {exclusion.shape[::-1]} differs from "
                         f"{Path(image_path).name} {image_shape[::-1]} (width, height)")
    return exclusion


def read_manifest(path: str | Path) -> pd.DataFrame:
    path = Path(path)
    headers = pd.read_csv(path, nrows=0, compression=None).columns
    columns = {c: c.strip().lower() for c in headers if c.strip().lower() in _MANIFEST_TYPES}
    if len(set(columns.values())) != len(columns):
        raise ValueError(f"manifest {path.name}: duplicate column names")
    m = pd.read_csv(path, usecols=list(columns), dtype={c: _MANIFEST_TYPES[key] for c, key in columns.items()},
                    compression=None).rename(columns=columns)
    missing = [c for c in REQUIRED if c not in m]
    if missing:
        raise ValueError(f"manifest {path.name}: missing column(s) {', '.join(missing)}")
    if m.empty:
        raise ValueError(f"manifest {path.name} has no rows")
    for c in REQUIRED:
        if m[c].isna().any():
            rows = ", ".join(str(i + 2) for i in m.index[m[c].isna()])
            raise ValueError(f"manifest {path.name}: empty '{c}' in line(s) {rows}")
    if _MANIFEST_ESCAPE in m:
        escaped = m.pop(_MANIFEST_ESCAPE).eq(_ESCAPE_VERSION).fillna(False)
        for column in ("image", "core", "name", "exclude", "units"):
            if column in m:
                m.loc[escaped, column] = m.loc[escaped, column].str.removeprefix("'")
    m["core"] = m["core"].fillna(path.stem) if "core" in m else path.stem
    m["units"] = m["units"].fillna("cm") if "units" in m else "cm"
    base = path.parent
    m["image"] = [str((base / p).resolve()) for p in m["image"]]
    if "exclude" in m:
        m["exclude"] = [str((base / p).resolve()) if isinstance(p, str) and p.strip() else None for p in m["exclude"]]
    else:
        m["exclude"] = None
    m["name"] = [n if isinstance(n, str) and n.strip() else Path(i).stem
                 for n, i in zip(m["name"] if "name" in m else [None] * len(m), m["image"])]
    return m


def write_manifest_template(folder: str | Path, out: str | Path | None = None) -> Path:
    """List the images of a folder in a manifest with empty depths, to be filled in by the user."""
    folder = Path(folder)
    imgs = sorted(p for p in folder.iterdir() if p.suffix.lower() in SUPPORTED and not p.stem.endswith("_excl"))
    if not imgs:
        raise ValueError(f"no images ({', '.join(sorted(SUPPORTED))}) in {folder}")
    out = Path(out) if out else folder / "sections.csv"
    rows = [{"image": p.name, "core": folder.name, "top": "", "bottom": "", "units": "cm", "name": p.stem,
             "exclude": (p.stem + "_excl.png") if (folder / (p.stem + "_excl.png")).exists() else ""} for p in imgs]
    for i, row in enumerate(rows):
        if any(spreadsheet_text(value) != value for value in row.values()):
            rows[i] = {key: "'" + value if value.startswith("'") else spreadsheet_text(value)
                       for key, value in row.items()} | {_MANIFEST_ESCAPE: _ESCAPE_VERSION}
    return write_csv_table(pd.DataFrame(rows), out)


def run_manifest(manifest: str | Path, to_mask: Callable[[Path], tuple] | None = None,
                 **analysis) -> dict[str, list[CoreSection]]:
    """Analyse every section of a manifest. Returns {core: [sections, top to bottom]}.

    to_mask(path) -> (mask, class_names[, exclusion]) turns an image into a label mask (default: an integer label mask, as
    exported by the program or by ``unet predict``). ``analysis`` goes to :func:`analyze_mask` (bin_cm, min_valid…).
    """
    m = read_manifest(manifest)
    to_mask = to_mask or (lambda p: (load_mask(p), None))
    cores: dict[str, list[CoreSection]] = {}
    for r in m.itertuples(index=False):
        mask, names, *rest = to_mask(Path(r.image))
        exclude = rest[0] if rest else None
        if isinstance(r.exclude, str):
            ex = _read_matching_exclusion(r.exclude, r.image, mask.shape[-2:])
            exclude = ex if exclude is None else (exclude | ex)
        s = analyze_mask(mask, float(r.top), float(r.bottom), units=r.units, class_names=names,
                         exclude=exclude, name=r.name, **analysis)
        s.data.insert(0, "core", r.core)
        s.metadata["core"] = r.core
        cores.setdefault(r.core, []).append(s)
    for secs in cores.values():
        secs.sort(key=lambda s: s.top * UNIT_TO_CM[s.units])
    return cores

from __future__ import annotations

def _safe(name: str) -> str:
    return "".join(c if c.isalnum() or c in "-_" else "_" for c in name) or "core"


def _batch_las_paths(out: Path, cores: Iterable[str]) -> dict[str, Path]:
    paths: dict[str, Path] = {}
    owners: dict[str, str] = {}
    for core in cores:
        path = out.with_name(f"{out.stem}_{_safe(core)}.las")
        key = normalize("NFC", path.name).casefold()
        if key in owners:
            raise ValueError(f"LAS output filename collision for cores {owners[key]!r} and {core!r} at {path.name!r}")
        owners[key] = core
        paths[core] = path
    for entry in out.parent.iterdir():
        if normalize("NFC", entry.name).casefold() in owners:
            raise FileExistsError(f"LAS output already exists at {entry}")
    return paths


def _parse_min_valid(text: str) -> float:
    try:
        return _validate_min_valid(float(text))
    except ValueError as error:
        raise argparse.ArgumentTypeError(str(error)) from error


def _mask_args(a: argparse.ArgumentParser) -> None:
    bins = a.add_mutually_exclusive_group()
    bins.add_argument("--bin-cm", type=float, default=None, help="interval thickness in cm (as in the program)")
    bins.add_argument("--bin-px", type=float, default=None, help="pixel rows per interval (default 1)")
    g = a.add_mutually_exclusive_group()
    g.add_argument("--threshold", type=float, default=5,
                   help="gray value <= threshold is bioturbation (default 5, v2.1 behaviour)")
    g.add_argument("--palette", type=Path, help="JSON {class: '#rrggbb'} for colour-coded images")
    g.add_argument("--label-mask", action="store_true", help="image is already an integer label mask")
    a.add_argument("--tolerance", type=int, default=10, help="colour tolerance for --palette")
    a.add_argument("--exclude-value", type=int, default=None,
                   help="with --label-mask: pixels with this value are excluded (e.g. 255 in U-Net datasets)")
    a.add_argument("--min-valid", type=_parse_min_valid, default=0.2,
                   help="minimum valid pixel fraction between 0 and 1 (default 0.2)")


def _mask_reader(args):
    """Function image path -> (mask, class names, exclusion or None) for the chosen input type."""
    palette = json.loads(args.palette.read_text(encoding="utf-8")) if args.palette else None
    if args.exclude_value is not None and not args.label_mask:
        raise SystemExit("--exclude-value needs --label-mask")

    def read(path: Path):
        if args.label_mask:
            mask = load_mask(path)
            if args.exclude_value is None:
                return mask, None, None
            ex = mask == args.exclude_value
            return np.where(ex, 0, mask), None, ex
        if palette is not None:
            return (*mask_from_palette(load_image(path, "RGB"), palette, args.tolerance), None)
        return (*mask_from_threshold(load_image(path), args.threshold), None)
    return read


def _bins(args) -> dict:
    return {"bin_cm": args.bin_cm, "bin_px": args.bin_px or 1, "min_valid": args.min_valid}


def _write(sections, out: Path, overlap: str | None, well: str = "") -> Path:
    ext = out.suffix.lower()
    if ext == ".las":
        return to_las(sections, out, well=well, overlap=overlap or "upper")
    if overlap and overlap != "keep":
        rows = join_sections(sections, overlap=overlap)
        return (write_excel_table if ext == ".xlsx" else write_csv_table)(rows, out)
    return (to_excel if ext == ".xlsx" else to_csv)(sections, out)

def main(argv=None) -> int:
    args = build_parser().parse_args(argv)
    if args.command == "analyze":
        mask, names, ex = _mask_reader(args)(args.image)
        if args.exclude:
            ex2 = _read_matching_exclusion(args.exclude, args.image, mask.shape[-2:])
            ex = ex2 if ex is None else (ex | ex2)
        section = analyze_mask(mask, args.top, args.bottom, units=args.units, class_names=names, exclude=ex,
                               name=args.name or args.image.stem, **_bins(args))
        out = args.output or args.image.with_name(args.image.stem + "_sedcoretool.csv")
        _write([section], out, None)
        print(f"{len(section.data)} data points -> {out}")
    elif args.command == "batch":
        read = _mask_reader(args)
        cores = run_manifest(args.manifest, to_mask=read, **_bins(args))
        out = args.output
        if out.suffix.lower() == ".las" and len(cores) > 1:
            for core, path in _batch_las_paths(out, cores).items():
                secs = cores[core]
                f = to_las(secs, path, well=core, overlap=args.overlap or "upper", overwrite=False)
                print(f"{core}: {len(secs)} sections -> {f}")
        else:
            secs = [s for v in cores.values() for s in v]
            _write(secs, out, args.overlap, next(iter(cores)))
            print(f"{len(cores)} core(s), {len(secs)} sections, {sum(len(s.data) for s in secs)} data points -> {out}")
    elif args.command == "manifest":
        print(f"fill in top and bottom -> {write_manifest_template(args.folder, args.output)}")
    elif args.command == "unet":
        return _unet(args)
    return 0


def _unet(args) -> int:
    if args.unet == "prepare":
        from .unet.dataset import prepare_dataset
        s = prepare_dataset(args.export, args.out, args.tile, args.overlap, args.px_per_cm, not args.multiclass,
                            args.val, args.test, args.seed, overwrite=args.overwrite)
        for k, v in s["split"].items():
            print(f"{k}: {v['tiles']} tiles, cores {', '.join(v['cores']) or '-'}")
        if not s["split"]["test"]["cores"]:
            print("warning: no test cores (fewer than 3 cores): scores will not measure independent cores")
    elif args.unet == "train":
        from .unet.train import train
        prog = (lambda d: print(json.dumps({"progress": d}), flush=True)) if args.progress_json else None
        s = train(args.data, args.out, args.encoder, None if args.no_pretrained else "imagenet", args.epochs,
                  args.batch, args.lr, args.patience, args.seed, args.device,
                  log=lambda m: print(m, flush=True), progress=prog, resume=args.resume, cpu_only=args.cpu_only)
        print(f"best epoch {s['bestEpoch']}, validation Dice {s['valDice']:.4f} -> {args.out.with_suffix('.onnx')}")
    elif args.unet == "device":
        from .unet.train import device_info
        print(json.dumps(device_info(cpu_only=args.cpu_only)))
    elif args.unet == "evaluate":
        from .unet.evaluate import evaluate
        r = evaluate(args.export, args.model, args.cores, args.threshold, args.bin_cm, args.dataset, cpu_only=args.cpu_only)
        p = r["pooled"]
        print(f"cores {', '.join(r['cores'])}: IoU {p['iou']:.4f}  Dice {p['dice']:.4f}")
        if args.output:
            args.output.write_text(json.dumps(r, indent=2, default=float), encoding="utf-8")
    elif args.unet == "predict":
        from PIL import Image

        from .segmentation import mask_from_unet
        mask, names = mask_from_unet(load_image(args.image, "RGB"), args.model, args.px_per_cm, args.threshold,
                                     cpu_only=args.cpu_only)
        out = args.output or args.image.with_name(args.image.stem + "_unet_mask.png")
        Image.fromarray(mask.astype("uint8")).save(out)
        print(f"classes {names}; {100 * (mask > 0).mean():.2f} % bioturbation -> {out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Depth and age handling."""

from __future__ import annotations

import numpy as np


def pixel_depths(n_rows: int, top: float, bottom: float) -> np.ndarray:
    """Depth at the centre of every pixel row (row 0 = top)."""
    if bottom <= top:
        raise ValueError("bottom depth must be greater than top depth")
    step = (bottom - top) / n_rows
    return top + (np.arange(n_rows) + 0.5) * step


def depth_to_age(depth: np.ndarray, model_depth: np.ndarray, model_age: np.ndarray,
                 extrapolate: bool = False) -> np.ndarray:
    """Linear interpolation of an age model (replaces v2.1 AgeConverter).

    Outside the model range returns NaN unless ``extrapolate`` is True.
    """
    md = np.asarray(model_depth, dtype=float)
    ma = np.asarray(model_age, dtype=float)
    if md.ndim != 1 or ma.ndim != 1 or len(md) < 2 or len(md) != len(ma) or not np.isfinite(md).all() or not np.isfinite(ma).all():
        raise ValueError("age model requires at least two finite depth-age pairs")
    order = np.lexsort((ma, md))
    md, ma = md[order], ma[order]
    keep = np.ones(len(md), dtype=bool)
    keep[1:] = (np.diff(md) != 0) | (np.diff(ma) != 0)  # identical repeated rows are merged
    md, ma = md[keep], ma[keep]
    if np.any(np.diff(md) == 0):
        raise ValueError("one depth has two different ages; write a hiatus as two depths a small distance apart")
    if len(md) < 2:
        raise ValueError("age model requires at least two finite depth-age pairs")
    d = np.asarray(depth, dtype=float)
    age = np.interp(d, md, ma)
    if extrapolate:
        lo = (ma[1] - ma[0]) / (md[1] - md[0])
        hi = (ma[-1] - ma[-2]) / (md[-1] - md[-2])
        age = np.where(d < md[0], ma[0] + (d - md[0]) * lo, age)
        age = np.where(d > md[-1], ma[-1] + (d - md[-1]) * hi, age)
    else:
        age = np.where((d < md[0]) | (d > md[-1]), np.nan, age)
    return age

"""Export results for correlation with other core data: CSV, Excel and LAS 2.0."""

from __future__ import annotations

import re
from collections.abc import Iterable
from csv import QUOTE_NONNUMERIC
from datetime import date
from pathlib import Path

import numpy as np
import pandas as pd

from .pipeline import UNIT_TO_CM, CoreSection

DEPTH_COLS = ("depth_top", "depth_bottom", "depth")
_FORMULA_START = re.compile(r"^[ \u00a0]*[\t\r\n]|^[\s\x00-\x1f\ufeff]*[=+@\-＝＋－＠]")


def combine(sections: Iterable[CoreSection]) -> pd.DataFrame:
    return pd.concat([s.data for s in sections], ignore_index=True)


def join_sections(sections: Iterable[CoreSection], units: str | None = None, overlap: str = "keep") -> pd.DataFrame:
    """Join the sections of one core into a single series sorted by depth (as the program's core column).

    Intervals are not recomputed. Depths are converted to ``units`` (default: those of the uppermost section).
    Gaps between sections are left as they are. Where sections overlap (a splice), ``overlap="keep"`` keeps every
    interval, ``"upper"`` keeps the upper section and drops the intervals of a lower section whose centre falls
    inside a section above, and ``"error"`` raises ValueError.
    """
    if overlap not in ("keep", "upper", "error"):
        raise ValueError("overlap must be 'keep', 'upper' or 'error'")
    secs = sorted(sections, key=lambda s: s.top * UNIT_TO_CM[s.units])
    if not secs:
        raise ValueError("no sections to join")
    units = units or secs[0].units
    parts, covered = [], []   # covered: (top, bottom) of the sections already added, in output units
    for s in secs:
        f = UNIT_TO_CM[s.units] / UNIT_TO_CM[units]
        d = s.data.copy()
        for c in DEPTH_COLS:
            if c in d:
                d[c] = d[c] * f
        top, bottom = s.top * f, s.bottom * f
        inside = np.zeros(len(d), dtype=bool)
        for a, b in covered:
            inside |= (d["depth"].to_numpy() > a) & (d["depth"].to_numpy() < b)
        if inside.any():
            if overlap == "error":
                raise ValueError(f"section {s.name!r} overlaps a section above it; use overlap='upper' or 'keep'")
            if overlap == "upper":
                d = d[~inside]
        covered.append((top, bottom))
        parts.append(d)
    out = pd.concat(parts, ignore_index=True).sort_values("depth", kind="stable", ignore_index=True)
    out.attrs["units"] = units
    return out

def _mnemonic(name: str, used: set[str]) -> str:
    m = re.sub(r"[^A-Za-z0-9_]", "_", name).strip("_").upper()[:24] or "CURVE"

def to_las(sections: Iterable[CoreSection], path: str | Path, well: str = "", units: str | None = None,
           overlap: str = "upper", *, overwrite: bool = True) -> Path:
    """LAS 2.0 file (one core) for well-log software. Depth = interval centre; no data = -999.25.

    The sections are joined with :func:`join_sections` (default ``overlap="upper"``, because depth must increase).
    STEP is the interval spacing when it is regular and 0 otherwise (gaps, mixed interval thickness).
    Set ``overwrite=False`` to create the destination exclusively.
    """
    from . import __version__
    sections = list(sections)
    df = join_sections(sections, units, overlap)
    units = df.attrs["units"]
    if df.empty:
        raise ValueError("no intervals to write")
    dep = df["depth"].to_numpy(dtype=float)
    if np.any(np.diff(dep) <= 0):
        raise ValueError("depths must increase strictly; overlapping sections need overlap='upper'")
    steps = np.diff(dep)
    step = float(np.round(steps[0], 6)) if len(steps) and np.allclose(steps, steps[0], rtol=1e-6, atol=1e-9) else 0.0
    du = _LAS_UNITS[units]
    used = {"DEPT"}
    curves = [("DEPT", du, "Depth of the interval centre", "depth")]
    curves += [(_mnemonic("DTOP", used), du, "Interval top", "depth_top"),
               (_mnemonic("DBOT", used), du, "Interval bottom", "depth_bottom")]
    if "age" in df:
        curves.append((_mnemonic("AGE", used), "", "Age from the age model", "age"))
    curves += [(_mnemonic("VALID", used), "%", "Valid (not excluded) pixels", "valid_pct"),
               (_mnemonic("BIOT", used), "%", "Bioturbation", "bioturbation_pct")]
    names = list(dict.fromkeys(n for s in sections for n in s.class_names))
    for n in names:
        if f"{n}_pct" in df:
            curves.append((_mnemonic(f"{n}_PCT", used), "%", f"{n} (percent of valid pixels)", f"{n}_pct"))
    for c in df.columns[df.columns.get_loc("bioturbation_pct") + 1:]:
        if c not in {col for *_, col in curves} and c not in ("section", "core", "age"):
            curves.append((_mnemonic(c, used), "", f"Index {c}", c))
    cols = [c for c in curves if c[3] in df]
    data = np.column_stack([pd.to_numeric(df[c[3]], errors="coerce").astype(float).to_numpy() for c in cols])
    data = np.where(np.isfinite(data), data, LAS_NULL)

    def line(mnem, unit, value, desc):
        return f"{mnem:<8}.{unit:<6} {value:<28}: {desc}"

    fmt = lambda x: f"{x:.4f}"  # noqa: E731
    out = ["~VERSION INFORMATION",
           line("VERS", "", "2.0", "CWLS LOG ASCII STANDARD - VERSION 2.0"),
           line("WRAP", "", "NO", "ONE LINE PER DEPTH STEP"),
           "~WELL INFORMATION",
           line("STRT", du, fmt(dep[0]), "START DEPTH"),
           line("STOP", du, fmt(dep[-1]), "STOP DEPTH"),
           line("STEP", du, fmt(step), "STEP (0 = IRREGULAR)"),
           line("NULL", "", str(LAS_NULL), "NULL VALUE"),
           line("COMP", "", "", "COMPANY"),
           line("WELL", "", well or sections[0].name, "CORE OR WELL"),
           line("FLD", "", "", "FIELD"),
           line("LOC", "", "", "LOCATION"),
           line("SRVC", "", f"sedcoretool {__version__}", "SERVICE COMPANY"),
           line("DATE", "", date.today().isoformat(), "DATE"),
           "~CURVE INFORMATION"]
    out += [line(m, u, "", d) for m, u, d, _ in cols]
    out += ["~PARAMETER INFORMATION",
            line("NSEC", "", str(len(sections)), "Number of core sections"),
            line("SECT", "", ", ".join(s.name for s in sorted(sections, key=lambda s: s.top * UNIT_TO_CM[s.units])),
                 "Sections, top to bottom")]
    out.append("~A " + " ".join(m for m, *_ in cols))
    out += [" ".join(f"{v:12.4f}" for v in row) for row in data]
    path = Path(path)
    with path.open("w" if overwrite else "x", encoding="ascii", errors="replace") as file:
        file.write("\n".join(out) + "\n")
    return path

"""Semi-quantitative bioturbation indices derived from percentage.

Schemes are data, so users can add or modify them without touching code.
Each tier is a half-open interval [lower, upper) in percent.
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import cached_property
from heapq import heappop, heappush

import numpy as np


@dataclass(frozen=True)
class IndexScheme:
    name: str
    tiers: tuple[tuple[float, float, int], ...]  # (lower, upper, value)
    reference: str = ""

    @cached_property
    def _partition(self) -> tuple[np.ndarray, np.ndarray]:
        # Sweep the bounds once. The last declared tier wins on overlaps.
        starts = sorted((lo, i, hi, value) for i, (lo, hi, value) in enumerate(self.tiers) if lo < hi)
        edges = np.array(sorted({edge for lo, _, hi, _ in starts for edge in (lo, hi)}), dtype=float)
        classes = np.full(edges.size + 1, -1, dtype=int)
        active: list[tuple[int, float, int]] = []
        next_start = 0
        for j, edge in enumerate(edges, 1):
            while next_start < len(starts) and starts[next_start][0] <= edge:
                _, order, hi, value = starts[next_start]
                heappush(active, (-order, hi, value))
                next_start += 1
            while active and active[0][1] <= edge:
                heappop(active)
            if active:
                classes[j] = active[0][2]
        return edges, classes

    @cached_property
    def _dense_lookup(self) -> np.ndarray | None:
        edges, classes = self._partition
        if not edges.size or edges[0] < 0 or edges[-1] > 256 or np.any(edges != np.floor(edges)):
            return None
        samples = np.arange(int(edges[-1]) + 1)
        return np.r_[classes[np.searchsorted(edges, samples, side="right")], -1]

    def classify(self, percentage: np.ndarray) -> np.ndarray:
        p = np.asarray(percentage, dtype=float)
        lookup = self._dense_lookup
        if lookup is None:
            edges, classes = self._partition
            return np.asarray(classes[np.searchsorted(edges, p, side="right")])

        # NaN and negative percentages use the sentinel at index -1.
        samples = np.empty_like(p)
        np.fmax(p, -1, out=samples)
        np.minimum(samples, lookup.size - 2, out=samples)
        np.floor(samples, out=samples)
        return np.asarray(lookup[samples.astype(np.intp)])

"""Image input. Images are oriented with the core top at row 0."""

from __future__ import annotations

from contextlib import contextmanager
from pathlib import Path
from struct import Struct
from zipfile import ZIP_DEFLATED, ZIP_STORED, ZipFile

import numpy as np
from PIL import Image, ImageFile, ImageMode

SUPPORTED = {".png", ".tif", ".tiff", ".jpg", ".jpeg", ".bmp"}
LOSSY = {".jpg", ".jpeg"}
_NATIVE_IMAGE_MODES = {"L", "RGB", "RGBA", "I", "I;16"}
# Covers a full-resolution IODP section (1,380 × 29,880) even with a four-byte output conversion.
_IMAGE_LOAD_BUDGET_BYTES = 768 * 1024**2


def _image_load_bytes(width: int, height: int, mode: str, *, convert: bool, mask: bool) -> int:
    pixels = width * height
    try:
        descriptor = ImageMode.getmode(mode)
    except KeyError:
        raise ValueError(f"Unsupported image output mode {mode!r}") from None
    packed = pixels * len(descriptor.bands) * np.dtype(descriptor.typestr).itemsize
    raster = 4 * pixels + 64 * height  # Pillow pixel storage plus row pointers and alignment allowance.
    decode_peak = 2 * raster  # Source and an allowance for codec staging.
    convert_peak = 3 * raster if convert else 0  # Source, intermediate and destination.
    array_peak = (2 if convert else 1) * raster + 2 * packed  # Pillow's raw chunks and their joined bytes.
    mask_peak = packed + 4 * pixels if mask else 0  # The packed array and its int32 copy after closing Pillow.
    return max(decode_peak, convert_peak, array_peak, mask_peak) + max(ImageFile.MAXBLOCK, 4 * width)


def load_image(path: str | Path, mode: str | None = None) -> np.ndarray:
    """Load an image as a numpy array.

    mode: optional PIL mode ("L" grayscale, "RGB"). None keeps the original.
    Lossy formats (JPEG) are accepted but discouraged for masks, because
    compression alters the pixel values the analysis relies on.
    Raises ValueError if the estimated per-image pixel buffers exceed 768 MiB.
    """
    return _load_pixels(path, mode, mask=False)


def _load_pixels(path: str | Path, mode: str | None, *, mask: bool) -> np.ndarray:
    path = Path(path)
    if path.suffix.lower() not in SUPPORTED:
        raise ValueError(f"Unsupported format {path.suffix!r}; use one of {sorted(SUPPORTED)}")
    with Image.open(path) as im:
        if mode is None and im.mode not in _NATIVE_IMAGE_MODES:
            mode = "RGB"
        estimate = _image_load_bytes(*im.size, im.mode if mode is None else mode, convert=mode is not None, mask=mask)
        if estimate > _IMAGE_LOAD_BUDGET_BYTES:
            raise ValueError(
                f"{path.name} ({im.width} × {im.height}): estimated image loading memory "
                f"({estimate:,} bytes) exceeds the {_IMAGE_LOAD_BUDGET_BYTES:,}-byte budget"
            )
        if mode is not None:
            with im.convert(mode) as converted:
                return np.asarray(converted)
        return np.asarray(im)


def load_mask(path: str | Path) -> np.ndarray:
    """Load a label mask (0 = background, 1..n = classes) saved as PNG/TIFF."""
    arr = _load_pixels(path, None, mask=True)
    if arr.ndim != 2:
        raise ValueError("A label mask must be single-channel (2-D). "
                         "For colour-coded masks use segmentation.mask_from_palette.")
    return arr.astype(np.int32)


def to_gray(img: np.ndarray) -> np.ndarray:
    """Luminance conversion (same weights as v2.1: 0.30, 0.56, 0.11)."""
    if img.ndim == 2:
        return img.astype(np.float64)
    rgb = img[..., :3].astype(np.float64)
    return rgb @ np.array([0.30, 0.56, 0.11])


# U-Net ZIPs contain images and a small manifest, all loaded into RAM by the reader.
_DATASET_ZIP_LIMIT = 2 * 1024**3
_DATASET_TOTAL_LIMIT = 1024**3
_DATASET_ENTRY_LIMIT = 256 * 1024**2
_DATASET_DIRECTORY_LIMIT = 16 * 1024**2
_DATASET_MANIFEST_LIMIT = 16 * 1024**2
_ZIP_END = Struct("<4s4H2IH")


def _dataset_entry_limit(name: str) -> int:
    return _DATASET_MANIFEST_LIMIT if name == "manifest.json" else _DATASET_ENTRY_LIMIT


def _check_dataset_zip_directory(file) -> int:
    size = file.seek(0, 2)
    if size > _DATASET_ZIP_LIMIT:
        raise ValueError("training ZIP exceeds the archive size limit")
    start = max(0, size - (65535 + _ZIP_END.size))
    file.seek(start)
    tail = file.read()
    pos = tail.rfind(b"PK\x05\x06")
    if pos < 0 or pos + _ZIP_END.size > len(tail):
        raise ValueError("invalid training ZIP directory")
    _, disk, dir_disk, disk_entries, count, length, offset, comment = _ZIP_END.unpack_from(tail, pos)
    if (pos + _ZIP_END.size + comment != len(tail) or disk or dir_disk or disk_entries != count or
            length > _DATASET_DIRECTORY_LIMIT or length < 46 * count or offset + length != start + pos):
        raise ValueError("invalid or oversized training ZIP directory")
    # Refuse ZIP64 here so zipfile cannot override the directory size checked above.
    if start + pos >= 20:
        file.seek(start + pos - 20)
        if file.read(4) == b"PK\x06\x07":
            raise ValueError("ZIP64 training archives are unsupported")
    return count


def _safe_zip_name(name: str) -> bool:
    return (not name.startswith("/") and "\\" not in name and "\x00" not in name and
            ":" not in name.split("/", 1)[0] and
            all(part not in ("", ".", "..") for part in name.removesuffix("/").split("/")))


@contextmanager
def open_training_export(path: str | Path):
    """Yield a bounded member reader for a U-Net ZIP or an unpacked training folder."""
    root = Path(path).resolve()
    if root.is_dir():
        consumed = 0

        def read_file(name):
            nonlocal consumed
            if not _safe_zip_name(name):
                raise ValueError("invalid training export member path")
            item = (root / name).resolve()
            if not item.is_relative_to(root):
                raise ValueError("training export member escapes its folder")
            if not item.is_file():
                raise ValueError(f"missing or invalid training export file: {name}")
            with item.open("rb") as file:
                size = file.seek(0, 2)
                if size > _dataset_entry_limit(name):
                    raise ValueError(f"training export member exceeds the size limit: {name}")
                consumed += size
                if consumed > _DATASET_TOTAL_LIMIT:
                    raise ValueError("training export reads exceed the size limit")
                file.seek(0)
                data = file.read(size + 1)
            if len(data) != size:
                raise ValueError(f"training export member size changed: {name}")
            return data

        yield read_file
        return

    with root.open("rb") as file:
        count = _check_dataset_zip_directory(file)
        with ZipFile(file) as archive:
            members = archive.infolist()
            if len(members) != count:
                raise ValueError("training ZIP directory entry count mismatch")
            entries, expanded = {}, 0
            for member in members:
                name = member.orig_filename
                if not _safe_zip_name(name) or name in entries or member.flag_bits & 0x41:
                    raise ValueError("invalid or duplicate training ZIP member")
                if member.compress_type not in (ZIP_STORED, ZIP_DEFLATED):
                    raise ValueError("unsupported training ZIP compression")
                if member.compress_type == ZIP_STORED and member.file_size != member.compress_size:
                    raise ValueError("invalid stored training ZIP member size")
                expanded += member.file_size
                if member.file_size > _dataset_entry_limit(name) or expanded > _DATASET_TOTAL_LIMIT:
                    raise ValueError("training ZIP uncompressed size exceeds the import limit")
                entries[name] = member

            consumed = 0

            def read_member(name):
                nonlocal consumed
                member = entries[name]
                consumed += member.file_size
                if consumed > _DATASET_TOTAL_LIMIT:
                    raise ValueError("training ZIP read exceeds the uncompressed size limit")
                with archive.open(member) as source:
                    # ZipExtFile enforces its declared size and checks the CRC while reading.
                    data = source.read(member.file_size + 1)
                if len(data) != member.file_size:
                    raise ValueError(f"training ZIP member size mismatch: {name}")
                return data

            yield read_member

"""Core analysis: label mask -> continuous series per depth bin."""

from __future__ import annotations

@dataclass
class CoreSection:
    """Result for one core section. ``data`` has one row per depth bin."""
    name: str
    top: float
    bottom: float
    units: str
    bin_px: int
    class_names: list[str]
    data: pd.DataFrame
    metadata: dict = field(default_factory=dict)

    def add_age(self, model_depth, model_age, extrapolate: bool = False) -> "CoreSection":
        self.data["age"] = depth_to_age(self.data["depth"], model_depth, model_age, extrapolate)
        return self


def _edge(j: int, bin_px: float) -> int:
    return int(np.floor(j * bin_px + 0.5))


def bin_edges(n: int, bin_px: float, keep_partial: bool = True) -> list[tuple[int, int]]:
    """Row ranges of the bins: bin j spans [floor(j*bin_px + 0.5), floor((j+1)*bin_px + 0.5)).

    Same rule as ``binRows`` in the web app (V0.1). bin_px may be fractional (interval in cm × px/cm);
    an integer bin_px gives the classic fixed-size bins. The last bin is clipped at n (dropped if
    keep_partial is False).
    """
    out, j = [], 0
    while _edge(j, bin_px) < n:
        end = _edge(j + 1, bin_px)
        if end > n and not keep_partial:
            break
        out.append((_edge(j, bin_px), min(n, end)))
        j += 1
    return out


def _bin_sums(values: np.ndarray, rng: list[tuple[int, int]]) -> np.ndarray:
    return np.array([values[a:b].sum(axis=0) for a, b in rng], dtype=float).reshape(len(rng), *values.shape[1:])


UNIT_TO_CM = {"mm": 0.1, "cm": 1.0, "m": 100.0}
_MAX_MASK_CLASSES = 255  # All positive IDs in an 8-bit label mask.


def _unit_factor(units: str) -> float:
    try:
        return UNIT_TO_CM[units]
    except KeyError:
        raise ValueError(f"units must be one of {sorted(UNIT_TO_CM)}, not {units!r}") from None


def _validate_min_valid(min_valid: float) -> float:
    if not 0 <= min_valid <= 1:
        raise ValueError("min_valid must be a finite fraction between 0 and 1 inclusive")
    return min_valid

def _label_row_counts(mask: np.ndarray, valid: np.ndarray, n_classes: int) -> np.ndarray:
    """Count row/class keys in blocks of <=65,536 pixels and <=64 rows."""
    rows, width = mask.shape
    classes = n_classes + 1
    counts = np.zeros((rows, classes), dtype=np.int64)
    block_rows = min(64, max(1, 65536 // max(1, width)))
    for row in range(0, rows, block_rows):
        end = min(rows, row + block_rows)
        offsets = np.arange(end - row, dtype=np.intp)[:, None] * classes
        for col in range(0, width, 65536):
            keys = mask[row:end, col:col + 65536].astype(np.intp) + offsets
            selected = keys[valid[row:end, col:col + 65536]]
            counts[row:end] += np.bincount(selected, minlength=(end - row) * classes).reshape(end - row, classes)
    return counts


def analyze_mask(mask: np.ndarray, top: float, bottom: float, *, units: str = "cm",
                 bin_px: float = 1, bin_cm: float | None = None, class_names: Sequence[str] | None = None,
                 schemes: Sequence[IndexScheme] = DEFAULT_SCHEMES, exclude: np.ndarray | None = None,
                 min_valid: float = 0.2, keep_partial: bool = True, name: str = "section_1") -> CoreSection:
    """Compute bioturbation percentage (total and per class) along depth.

    mask: 2-D integer or boolean array, 0 = matrix, 1..255 = bioturbation classes; row 0 = top.
    Missing class IDs retain their positions. Class names and overlap layers are limited to 255.
    bin_px: number of pixel rows averaged into each data point (may be fractional).
    bin_cm: interval thickness in cm; if given, bin_px = bin_cm × pixel rows per cm (as in the web app).
    exclude: optional 2-D boolean array of the same size; True = excluded pixel (crack, gap, sample), as the
    program's exclusion layer. Percentages are 100 · Σ bioturbated / Σ valid pixels of each interval.
    min_valid: finite fraction in [0, 1]. An interval below this fraction of valid pixels has no data
    (NaN percentages and empty indices).

    Overlapping traces (V0.20, as in the web app): mask may instead be a 3-D boolean stack
    (n_classes, rows, cols), one layer per class. A pixel is bioturbated once if any layer is set, and
    counts for every class whose layer is set, so the class percentages may add up to more than the total.
    """
    _validate_min_valid(min_valid)
    mask = np.asarray(mask)
    if mask.ndim == 3:
        n_classes = mask.shape[0]
    elif mask.ndim == 2:
        if mask.dtype.kind not in "biu":
            raise ValueError("a 2-D mask must contain integer or boolean labels")
        if mask.size and mask.dtype.kind == "i" and mask.min() < 0:
            raise ValueError("mask labels must be non-negative")
        n_classes = int(mask.max()) if mask.size else 0
    else:
        raise ValueError("mask must be 2-D, or 3-D (classes, rows, cols) for overlapping traces")
    n_names = len(class_names) if class_names is not None else 0
    class_count = max(n_classes, n_names)
    if class_count > _MAX_MASK_CLASSES:
        raise ValueError(f"mask class count or maximum label {class_count} exceeds the limit of {_MAX_MASK_CLASSES}")
    if 0 < n_names < n_classes:
        raise ValueError("fewer class_names than classes present in the mask")

    layers = None
    if mask.ndim == 3:
        layers = mask.astype(bool)
        mask = layers.any(axis=0).astype(np.int32)
    if bottom <= top:
        raise ValueError("bottom depth must be greater than top depth")
    if bin_cm is not None:
        if bin_cm <= 0:
            raise ValueError("bin_cm must be positive")
        # Match app.js::pxPerCm/binRowsOf: divide first, then multiply. Reassociation
        # can move a half-row boundary and change percentages and indices.
        px_per_cm = mask.shape[0] / ((bottom - top) * _unit_factor(units))
        bin_px = bin_cm * px_per_cm
    if bin_px < 1:
        raise ValueError("bin_px must be >= 1 (interval thinner than one pixel row)")
    n_rows, n_cols = mask.shape
    valid = np.ones(mask.shape, dtype=bool)
    if exclude is not None:
        exclude = np.asarray(exclude, dtype=bool)
        if exclude.shape != mask.shape:
            raise ValueError(f"exclusion layer {exclude.shape} does not match the mask {mask.shape}")
        valid = ~exclude
    names = list(class_names) if n_names else [f"class_{i}" for i in range(1, n_classes + 1)]

    rng = bin_edges(n_rows, bin_px, keep_partial)
    r0 = np.array([a for a, _ in rng], dtype=float)
    r1 = np.array([b for _, b in rng], dtype=float)
    step = (bottom - top) / n_rows
    histogram = (_label_row_counts(mask, valid, len(names))
                 if layers is None and os.environ.get('SEDCORETOOL_EXPERIMENTAL_COUNTS') == '1' else None)
    v = _bin_sums(valid.sum(axis=1), rng)
    ok = (v > 0) & (v >= min_valid * (r1 - r0) * n_cols)
    with np.errstate(invalid="ignore", divide="ignore"):
        def pct(hit: np.ndarray) -> np.ndarray:
            return np.where(ok, 100 * _bin_sums((hit & valid).sum(axis=1), rng) / v, np.nan)

        def histogram_pct(values):
            return np.where(ok, 100 * _bin_sums(values, rng) / v, np.nan)

        cols = {
            "depth_top": top + r0 * step,
            "depth_bottom": top + r1 * step,
            "depth": top + (r0 + r1) / 2 * step,
            "valid_pct": 100 * v / ((r1 - r0) * n_cols),
            "bioturbation_pct": (pct(mask > 0) if histogram is None else histogram_pct(histogram[:, 1:].sum(axis=1))),
        }
        if len(names) > 1:
            for label, cname in enumerate(names, start=1):
                cols[f"{cname}_pct"] = (pct(layers[label - 1] if layers is not None else mask == label)
                                        if histogram is None else histogram_pct(histogram[:, label]))
    df = pd.DataFrame(cols)
    for scheme in schemes:
        idx = pd.Series(scheme.classify(df["bioturbation_pct"].to_numpy()), dtype="Int64")
        df[scheme.name] = idx.mask(idx < 0)   # no data -> empty, never 0
    df.insert(0, "section", name)

    return CoreSection(name=name, top=top, bottom=bottom, units=units, bin_px=bin_px,
                       class_names=names, data=df,
                       metadata={"image_rows": n_rows, "image_cols": n_cols,
                                 "px_per_unit": n_rows / (bottom - top),
                                 "excluded_pct": float(100 * (~valid).mean()) if valid.size else 0.0})


def analyze_image(path, top: float, bottom: float, *, threshold: float = 5, **kwargs) -> CoreSection:
    """v2.1-compatible entry point: treated image with bioturbation painted black."""
    mask, names = mask_from_threshold(load_image(path), threshold)
    return analyze_mask(mask, top, bottom, class_names=names, **kwargs)

"""Building label masks from treated images.

A label mask is a 2-D integer array: 0 = non-bioturbated matrix,
1..n = bioturbation classes (e.g. ichnotaxa).

Current methods reproduce the published workflow (features painted on
the image). Future learned segmenters (e.g. U-Net) should expose the
same output: ``(mask, class_names)``.
"""

from __future__ import annotations

from collections.abc import Mapping, Sequence

import numpy as np

from .io import to_gray


def mask_from_threshold(img: np.ndarray, threshold: float = 5,
                        class_name: str = "bioturbation") -> tuple[np.ndarray, list[str]]:
    """Binary mask: pixels with gray value <= threshold are bioturbation.

    This is the v2.1 behaviour (features painted black, threshold 5).
    """
    mask = (to_gray(img) <= threshold).astype(np.int32)
    return mask, [class_name]


def _hex_to_rgb(color: str | Sequence[int]) -> np.ndarray:
    if isinstance(color, str):
        c = color.lstrip("#")
        return np.array([int(c[i:i + 2], 16) for i in (0, 2, 4)], dtype=np.int32)
    return np.asarray(color[:3], dtype=np.int32)


def mask_from_palette(img: np.ndarray, palette: Mapping[str, str | Sequence[int]],
                      tolerance: int = 10) -> tuple[np.ndarray, list[str]]:
    """Multi-class mask from a colour-coded image.

    palette: {"Zoophycos": "#ff0000", "Chondrites": (0, 0, 255), ...}
    A pixel belongs to a class when every RGB channel is within
    ``tolerance`` of the class colour. Unmatched pixels are background.
    """
    if img.ndim != 3:
        raise ValueError("Palette masks need an RGB image.")
    rgb = img[..., :3].astype(np.int32)
    mask = np.zeros(rgb.shape[:2], dtype=np.int32)
    names = list(palette)
    for label, name in enumerate(names, start=1):
        ref = _hex_to_rgb(palette[name])
        hit = np.all(np.abs(rgb - ref) <= tolerance, axis=-1) & (mask == 0)
        mask[hit] = label
    return mask, names


def mask_from_unet(img: np.ndarray, model, px_per_cm: float | None = None,
                   threshold: float = 0.5, *, cpu_only: bool = False) -> tuple[np.ndarray, list[str]]:
    """Label mask from a trained U-Net (V0.15).

    ``model``: path to the .onnx file (its .json next to it) or an ``sedcoretool.unet.predict.OnnxUnet``.
    ``px_per_cm`` of the image, so it is resampled to the training resolution. Pixels with probability of
    bioturbation >= ``threshold`` get class 1 (binary model) or the most probable class (multi-class model).
    ``cpu_only`` enforces CPU execution even for a caller-configured ONNX session.
    """
    from .unet.predict import OnnxUnet

    net = model if isinstance(model, OnnxUnet) else OnnxUnet(model)
    rgb = img if img.ndim == 3 else np.repeat(img[..., None], 3, axis=2)
    kwargs = {"cpu_only": True} if cpu_only else {}
    prob, cls = net.predict(rgb[..., :3].astype(np.uint8), px_per_cm, **kwargs)
    on = prob >= threshold
    mask = np.where(on, cls if cls is not None else 1, 0).astype(np.int32)
    return mask, list(net.config.classes)

"""Training data for the U-Net (V0.15).

Input: the ZIP (or its unpacked folder) exported from the app with *Results → U-Net training data*:
``manifest.json`` + ``images/*.png`` (original analysis area) + ``masks/*.png`` (0 = matrix or artefact,
k = class k, 255 = ignore: excluded zones and traces without a confirmed class).

``prepare_dataset`` resamples every image to a common resolution (px/cm), cuts overlapping tiles and
splits them **by core**: all tiles of a core go to the same set, so the test score measures performance
on cores the model has never seen.
"""

from __future__ import annotations

import hashlib
import io
import json
import re
import shutil
from dataclasses import dataclass
from pathlib import Path

import numpy as np
from PIL import Image

from ..io import open_training_export
from .tiling import IGNORE, resample_bilinear, tile_starts


@dataclass
class Item:
    core: str
    section: str
    image: np.ndarray  # (h, w, 3) uint8
    mask: np.ndarray   # (h, w) uint8
    px_per_cm: float



def load_export(src: str | Path) -> tuple[list[Item], list[dict]]:
    """Items and classes of an exported training set (ZIP file or folder)."""
    with open_training_export(src) as read:
        man = json.loads(read("manifest.json"))
        if man.get("kind") != "segmentation-dataset":
            raise ValueError("not a SedCoreTool (or CorAnTool) segmentation dataset (manifest kind)")
        items, decoded = [], 0
        for it in man["items"]:
            with Image.open(io.BytesIO(read(it["image"]))) as image, Image.open(io.BytesIO(read(it["mask"]))) as mask:
                if image.size != mask.size:
                    raise ValueError(f"image and mask sizes differ: {it['image']}")
                pixels = image.width * image.height
                decoded += 4 * pixels  # RGB plus the grayscale mask
                if 4 * pixels > 256 * 1024**2 or decoded > 1024**3:
                    raise ValueError("decoded training images exceed the import limit")
                img, msk = np.asarray(image.convert("RGB")), np.asarray(mask.convert("L"))
            items.append(Item(it["core"], it["section"], img, msk, float(it["pxPerCm"])))
    return items, man["classes"]


def resample_mask(mask: np.ndarray, out_h: int, out_w: int) -> np.ndarray:
    """Nearest-neighbour resampling (labels must not be averaged), same pixel-centre rule as the images."""
    h, w = mask.shape
    ys = np.clip(np.floor((np.arange(out_h) + 0.5) * h / out_h), 0, h - 1).astype(int)
    xs = np.clip(np.floor((np.arange(out_w) + 0.5) * w / out_w), 0, w - 1).astype(int)
    return mask[ys][:, xs]


def binary_mask(mask: np.ndarray) -> np.ndarray:
    """All classes merged into bioturbation (1); matrix 0; ignore kept."""
    out = (mask > 0).astype(np.uint8)
    out[mask == IGNORE] = IGNORE
    return out


def split_cores(cores: list[str], val: list[str] | None = None, test: list[str] | None = None,
                val_frac: float = 0.2, test_frac: float = 0.2, seed: int = 42) -> dict[str, str]:
    """Core -> 'train' | 'val' | 'test'. Named cores are used as given; the rest are shuffled with ``seed``."""
    val, test = list(val or []), list(test or [])
    if set(val) & set(test):
        raise ValueError("validation and test cores must be disjoint")
    unknown = [c for c in val + test if c not in cores]
    if unknown:
        raise ValueError(f"unknown cores: {unknown}")
    rest = sorted(c for c in set(cores) if c not in val and c not in test)
    rng = np.random.default_rng(seed)
    rng.shuffle(rest)
    n = len(set(cores))
    if not test and n >= 3:
        test = rest[:max(1, round(test_frac * n))]
        rest = rest[len(test):]
    if not val and len(rest) >= 2:
        val = rest[:max(1, round(val_frac * n))]
        rest = rest[len(val):]
    out = {c: "train" for c in rest}
    out.update({c: "val" for c in val})
    out.update({c: "test" for c in test})
    return out


def tiles_of(item: Item, tile: int, overlap: int, px_per_cm: float | None, min_valid: float = 0.5):
    """Yield (y0, x0, image tile, mask tile) for tiles with at least ``min_valid`` non-ignored pixels."""
    img, msk = item.image, item.mask
    if px_per_cm:
        s = px_per_cm / item.px_per_cm
        if abs(s - 1) > 0.02:
            H, W = max(1, round(img.shape[0] * s)), max(1, round(img.shape[1] * s))
            img, msk = resample_bilinear(img, H, W), resample_mask(msk, H, W)
    h, w = msk.shape
    for y0 in tile_starts(h, tile, overlap):
        for x0 in tile_starts(w, tile, overlap):
            ys = np.minimum(h - 1, y0 + np.arange(tile))
            xs = np.minimum(w - 1, x0 + np.arange(tile))
            m = msk[ys][:, xs].copy()
            # padding outside the image is ignored, not matrix
            m[(y0 + np.arange(tile)) >= h, :] = IGNORE
            m[:, (x0 + np.arange(tile)) >= w] = IGNORE
            if (m != IGNORE).mean() < min_valid:
                continue
            yield y0, x0, img[ys][:, xs], m


def _tile_stems(items: list[Item]) -> list[str]:
    """Portable, bounded names whose identity does not depend on sanitised display labels."""
    stems, used = [], set()
    for item in items:
        labels = []
        for field, value in (("core", item.core), ("section", item.section)):
            if not isinstance(value, str) or not value:
                raise ValueError(f"invalid {field} identifier: expected a non-empty string")
            labels.append(re.sub(r"[^A-Za-z0-9_-]+", "_", value)[:32])
        identity = json.dumps([item.core, item.section], ensure_ascii=True, separators=(",", ":")).encode("ascii")
        stem = "__".join([*labels, hashlib.sha256(identity).hexdigest()])
        if stem.casefold() in used:
            raise ValueError(f"duplicate core/section identifiers: {item.core!r}, {item.section!r}")
        used.add(stem.casefold())
        stems.append(stem)
    return stems


def _dataset_path(output: Path, *parts: str) -> Path:
    """Reject aliases as well as escapes, before writing or replacing dataset contents."""
    path = output.joinpath(*parts)
    resolved = path.resolve()
    if not resolved.is_relative_to(output) or resolved != path:
        raise ValueError(f"unsafe dataset output path: {path}")
    return path


def prepare_dataset(src: str | Path, out: str | Path, tile: int = 512, overlap: int = 64,
                    px_per_cm: float | None = None, binary: bool = True, val_cores=None, test_cores=None,
                    seed: int = 42, min_valid: float = 0.5, overwrite: bool = False) -> dict:
    """Write ``out/{train,val,test}/{images,masks}/*.png`` and ``out/dataset.json``; return the summary.

    ``out`` must be empty, so that tiles of two datasets are never mixed. With ``overwrite=True`` a previous dataset
    in ``out`` (recognised by its ``dataset.json``) is replaced; any other non-empty directory is refused.
    Tile names use portable labels and the exact core/section identity. Duplicate identities and aliased output
    paths are refused before replacement; existing files are never silently overwritten during tile creation.
    """
    items, classes = load_export(src)
    if not items:
        raise ValueError("the export has no images")
    stems = _tile_stems(items)
    if px_per_cm is None:
        px_per_cm = float(np.median([it.px_per_cm for it in items]))
    split = split_cores([it.core for it in items], val_cores, test_cores, seed=seed)
    out = Path(out).resolve()
    partitions = {part: _dataset_path(out, part) for part in ("train", "val", "test")}
    directories = {(part, kind): _dataset_path(out, part, kind) for part in partitions for kind in ("images", "masks")}
    metadata = _dataset_path(out, "dataset.json")
    if out.exists() and any(out.iterdir()):
        if not overwrite:
            raise ValueError("output directory must be empty; choose a new directory or use --overwrite to replace "
                             "a previous dataset")
        if not metadata.is_file():
            raise ValueError("--overwrite only replaces a previous dataset (a directory containing dataset.json)")
        for directory in partitions.values():
            shutil.rmtree(directory, ignore_errors=True)
        metadata.unlink()
    counts = {k: {"tiles": 0, "cores": sorted({c for c, v in split.items() if v == k})} for k in ("train", "val", "test")}
    for directory in directories.values():
        directory.mkdir(parents=True, exist_ok=True)
    for it, stem in zip(items, stems, strict=True):
        part = split[it.core]
        for y0, x0, im, m in tiles_of(it, tile, overlap, px_per_cm, min_valid):
            if binary:
                m = binary_mask(m)
            name = f"{stem}__y{y0}_x{x0}.png"
            image_path = _dataset_path(out, part, "images", name)
            mask_path = _dataset_path(out, part, "masks", name)
            with image_path.open("xb") as image_file:
                Image.fromarray(im).save(image_file, format="PNG")
            with mask_path.open("xb") as mask_file:
                Image.fromarray(m).save(mask_file, format="PNG")
            counts[part]["tiles"] += 1
    names = ["bioturbation"] if binary else [c["name"] for c in classes]
    summary = {"tile": tile, "overlap": overlap, "pxPerCm": px_per_cm, "binary": binary, "classes": names,
               "ignore": IGNORE, "seed": seed, "split": counts}
    with metadata.open("x", encoding="utf-8") as file:
        file.write(json.dumps(summary, indent=2))
    return summary

"""Scoring a trained U-Net on whole images of independent cores (V0.15).

Tiles are only for training: evaluation predicts each complete analysis area, as the app does, and
compares it with the expert mask (pixels labelled 255 are ignored). Scores are given per image and
pooled (TP, FP and FN summed over all images before computing IoU and Dice).
"""

from __future__ import annotations

import json
from pathlib import Path

import numpy as np

from .dataset import load_export
from .metrics import pixel_scores, score
from .predict import OnnxUnet


def evaluate(export: str | Path, model: str | Path, cores: list[str] | None = None, threshold: float = 0.5,
             bin_cm: float = 1.0, dataset: str | Path | None = None, *, cpu_only: bool = False) -> dict:
    """Scores on the images of ``cores`` (default: the test cores recorded in ``dataset``/dataset.json)."""
    if cores is None and dataset is not None:
        cores = json.loads((Path(dataset) / "dataset.json").read_text(encoding="utf-8"))["split"]["test"]["cores"]
    items, _ = load_export(export)
    if cores is not None:
        items = [it for it in items if it.core in set(cores)]
    if not items:
        raise ValueError("no images to evaluate (check the core names)")
    net = OnnxUnet(model)
    rows, tp = [], np.zeros(3, int)
    for it in items:
        prob, _ = net.predict(it.image, it.px_per_cm, cpu_only=cpu_only)
        pred = (prob >= threshold).astype(np.uint8)
        s = score(pred, it.mask, bin_cm * it.px_per_cm)
        tp += [s["tp"], s["fp"], s["fn"]]
        rows.append({"core": it.core, "section": it.section, **s})
    t, f, n = tp
    pooled = {"tp": int(t), "fp": int(f), "fn": int(n), "iou": t / (t + f + n) if t + f + n else float("nan"),
              "dice": 2 * t / (2 * t + f + n) if t + f + n else float("nan")}
    return {"threshold": threshold, "bin_cm": bin_cm, "cores": sorted({r["core"] for r in rows}), "pooled": pooled, "images": rows}

"""Scores of a predicted mask against an expert mask (V0.15). Pixels equal to 255 in the truth are ignored.

Pixel scores (bioturbation = any class > 0): IoU = TP / (TP + FP + FN), Dice = 2TP / (2TP + FP + FN),
precision = TP / (TP + FP), recall = TP / (TP + FN).
Per bin of ``bin_px`` rows (same bins as ``analyze_mask``): bioturbation % = 100 · positives / valid pixels;
reported as mean absolute error (percentage points), bias (prediction − truth) and the share of bins whose
BI (Taylor & Goldring) and ii (Droser & Bottjer) class is the same.
"""

from __future__ import annotations

import numpy as np

from ..indices import BI_TAYLOR_GOLDRING, II_DROSER_BOTTJER
from ..pipeline import bin_edges
from .tiling import IGNORE


def pixel_scores(pred: np.ndarray, truth: np.ndarray) -> dict:
    valid = truth != IGNORE
    p, t = (pred > 0) & valid, (truth > 0) & valid
    tp, fp, fn = int((p & t).sum()), int((p & ~t).sum()), int((~p & t & valid).sum())
    div = lambda a, b: a / b if b else float("nan")
    return {"tp": tp, "fp": fp, "fn": fn, "valid": int(valid.sum()), "iou": div(tp, tp + fp + fn),
            "dice": div(2 * tp, 2 * tp + fp + fn), "precision": div(tp, tp + fp), "recall": div(tp, tp + fn)}


def bin_percentages(mask: np.ndarray, truth: np.ndarray, bin_px: float) -> np.ndarray:
    """Bioturbation % of ``mask`` in each bin of rows, counting only pixels that are valid in ``truth``."""
    valid = truth != IGNORE
    pos = ((mask > 0) & valid).sum(axis=1).astype(float)
    n = valid.sum(axis=1).astype(float)
    out = []
    for a, b in bin_edges(mask.shape[0], bin_px):
        v = n[a:b].sum()
        out.append(100 * pos[a:b].sum() / v if v else np.nan)
    return np.array(out)


def bin_scores(pred: np.ndarray, truth: np.ndarray, bin_px: float) -> dict:
    pp, tp = bin_percentages(pred, truth, bin_px), bin_percentages(truth, truth, bin_px)
    ok = ~np.isnan(tp)
    pp, tp = pp[ok], tp[ok]
    if not len(tp):
        return {"bins": 0}
    return {"bins": int(len(tp)), "mae_pct": float(np.abs(pp - tp).mean()), "bias_pct": float((pp - tp).mean()),
            "bi_agreement": float((BI_TAYLOR_GOLDRING.classify(pp) == BI_TAYLOR_GOLDRING.classify(tp)).mean()),
            "ii_agreement": float((II_DROSER_BOTTJER.classify(pp) == II_DROSER_BOTTJER.classify(tp)).mean())}


def score(pred: np.ndarray, truth: np.ndarray, bin_px: float) -> dict:
    return {**pixel_scores(pred, truth), **bin_scores(pred, truth, bin_px)}

"""Running a trained U-Net (.onnx + .json) with ONNX Runtime, with the same tiling as the app (V0.15)."""

from __future__ import annotations

import json
from pathlib import Path

import numpy as np

from .tiling import UnetConfig, predict_proba


class OnnxUnet:
    """``OnnxUnet("model.onnx")`` reads ``model.json`` next to it (tile, overlap, normalisation, px/cm, classes)."""

    def __init__(self, path: str | Path, config: UnetConfig | None = None):
        try:
            import onnxruntime as ort
        except ImportError as e:  # pragma: no cover - optional dependency
            raise ImportError("install the optional dependencies: pip install 'sedcoretool[unet]'") from e
        path = self._path = Path(path)
        js = path.with_suffix(".json")
        self.config = config or (UnetConfig.from_json(json.loads(js.read_text(encoding="utf-8"))) if js.exists() else UnetConfig())
        self.session = ort.InferenceSession(str(path), providers=["CPUExecutionProvider"])
        self._in = self.session.get_inputs()[0].name

    def run_tile(self, x: np.ndarray) -> np.ndarray:
        return self.session.run(None, {self._in: x[None].astype(np.float32)})[0][0]

    def predict(self, rgb: np.ndarray, px_per_cm: float | None = None, *, cpu_only: bool = False):
        """(probability of bioturbation, class map or None) for an (h, w, 3) uint8 image."""
        scale = self.config.px_per_cm / px_per_cm if (self.config.px_per_cm and px_per_cm) else 1.0
        net = self
        if cpu_only and self.session.get_providers() != ["CPUExecutionProvider"]:
            # Keep any caller-configured provider session intact.
            net = OnnxUnet(self._path, self.config)
        return predict_proba(net.run_tile, rgb, self.config, scale)

"""Tiling, normalisation, blending and resampling for the U-Net (V0.15).

Identical rules to ``app/js/unet.js``; tests in both languages check the same values:

* tile starts along an axis: step ``tile - overlap``, last tile flush with the end, one tile if ``len <= tile``;
* edge padding (coordinates clamped) for images smaller than a tile;
* blending weight per axis ``min(1, (i + 0.5) / overlap, (tile - i - 0.5) / overlap)``, 2-D weight = product;
* bilinear resampling with aligned pixel centres ``xs = (x + 0.5) * w / W - 0.5`` (clamped);
* sigmoid (one output channel) or softmax (several; channel 0 = matrix) of logits.
"""

from __future__ import annotations

from collections.abc import Callable, Sequence
from dataclasses import dataclass, field

import numpy as np

IGNORE = 255


@dataclass
class UnetConfig:
    """Description of a trained model (the JSON written next to the .onnx file)."""
    tile: int = 512
    overlap: int = 64
    mean: Sequence[float] = (0.485, 0.456, 0.406)
    std: Sequence[float] = (0.229, 0.224, 0.225)
    px_per_cm: float | None = None
    classes: list[str] = field(default_factory=lambda: ["bioturbation"])
    output: str = "logits"  # or "probs"

    def to_json(self) -> dict:
        return {"tile": self.tile, "overlap": self.overlap, "mean": list(self.mean), "std": list(self.std),
                "pxPerCm": self.px_per_cm, "classes": list(self.classes), "output": self.output}

    @classmethod
    def from_json(cls, d: dict) -> "UnetConfig":
        return cls(tile=d.get("tile", 512), overlap=d.get("overlap", 64), mean=d.get("mean", cls.mean),
                   std=d.get("std", cls.std), px_per_cm=d.get("pxPerCm"),
                   classes=d.get("classes", ["bioturbation"]), output=d.get("output", "logits"))


def tile_starts(length: int, tile: int, overlap: int) -> list[int]:
    if length <= tile:
        return [0]
    step = max(1, tile - overlap)
    out, s = [], 0
    while s + tile < length:
        out.append(s)
        s += step
    if out[-1] != length - tile:
        out.append(length - tile)
    return out


def blend_window(tile: int, overlap: int) -> np.ndarray:
    i = np.arange(tile, dtype=np.float64)
    if overlap <= 0:
        return np.ones(tile)
    return np.minimum(1.0, np.minimum((i + 0.5) / overlap, (tile - i - 0.5) / overlap))


def resample_bilinear(img: np.ndarray, out_h: int, out_w: int) -> np.ndarray:
    """Bilinear resampling of an (h, w) or (h, w, c) array; uint8 input is rounded half to even and clipped."""
    h, w = img.shape[:2]
    a = img.astype(np.float64)
    sy, sx = h / out_h, w / out_w  # same operation order as the JS version (exact uint8 parity)
    fy = np.clip((np.arange(out_h) + 0.5) * sy - 0.5, 0, h - 1)
    fx = np.clip((np.arange(out_w) + 0.5) * sx - 0.5, 0, w - 1)
    y0, x0 = np.floor(fy).astype(int), np.floor(fx).astype(int)
    y1, x1 = np.minimum(h - 1, y0 + 1), np.minimum(w - 1, x0 + 1)
    ty, tx = fy - y0, fx - x0
    if a.ndim == 3:
        ty, tx = ty[:, None, None], tx[None, :, None]
    else:
        ty, tx = ty[:, None], tx[None, :]
    top = a[y0][:, x0] * (1 - tx) + a[y0][:, x1] * tx
    bot = a[y1][:, x0] * (1 - tx) + a[y1][:, x1] * tx
    out = top * (1 - ty) + bot * ty
    if img.dtype == np.uint8:
        return np.clip(np.rint(out), 0, 255).astype(np.uint8)
    return out.astype(img.dtype if np.issubdtype(img.dtype, np.floating) else np.float64)


def tile_input(rgb: np.ndarray, x0: int, y0: int, tile: int, mean, std) -> np.ndarray:
    """(3, tile, tile) float32 input for the tile at (x0, y0) of an (h, w, 3) uint8 image, edge padded."""
    h, w = rgb.shape[:2]
    ys = np.minimum(h - 1, y0 + np.arange(tile))
    xs = np.minimum(w - 1, x0 + np.arange(tile))
    t = rgb[ys][:, xs, :3].astype(np.float32) / 255.0
    t = (t - np.asarray(mean, np.float32)) / np.asarray(std, np.float32)
    return np.ascontiguousarray(t.transpose(2, 0, 1))


def to_probs(out: np.ndarray, logits: bool = True) -> np.ndarray:
    """(C, H, W) network output to probabilities."""
    out = out.astype(np.float64)
    if not logits:
        return out
    if out.shape[0] == 1:
        return 1.0 / (1.0 + np.exp(-out))
    e = np.exp(out - out.max(axis=0, keepdims=True))
    return e / e.sum(axis=0, keepdims=True)


def predict_proba(run: Callable[[np.ndarray], np.ndarray], rgb: np.ndarray, cfg: UnetConfig,
                  scale: float = 1.0) -> tuple[np.ndarray, np.ndarray | None]:
    """Probability of bioturbation (h, w) and, for several classes, argmax class 1..C-1 (else None).

    ``run`` takes a (3, tile, tile) float32 array and returns the (C, tile, tile) output.
    ``scale`` = model px/cm / image px/cm (the image is resampled when it differs from 1 by more than 2 %).
    """
    h, w = rgb.shape[:2]
    img, H, W = rgb, h, w
    if abs(scale - 1) > 0.02:
        H, W = max(1, round(h * scale)), max(1, round(w * scale))
        img = resample_bilinear(rgb[..., :3], H, W)
    T = cfg.tile
    xs, ys, win = tile_starts(W, T, cfg.overlap), tile_starts(H, T, cfg.overlap), blend_window(T, cfg.overlap)
    w2 = np.outer(win, win)
    acc, wsum = None, np.zeros((H, W))
    for y0 in ys:
        for x0 in xs:
            p = to_probs(np.asarray(run(tile_input(img, x0, y0, T, cfg.mean, cfg.std))), cfg.output != "probs")
            if acc is None:
                acc = np.zeros((p.shape[0], H, W))
            hh, ww = min(T, H - y0), min(T, W - x0)
            acc[:, y0:y0 + hh, x0:x0 + ww] += w2[:hh, :ww] * p[:, :hh, :ww]
            wsum[y0:y0 + hh, x0:x0 + ww] += w2[:hh, :ww]
    acc /= np.where(wsum > 0, wsum, 1)
    planes = acc if (H, W) == (h, w) else resample_bilinear(acc.transpose(1, 2, 0), h, w).transpose(2, 0, 1)
    if planes.shape[0] == 1:
        return planes[0], None
    cls = 1 + np.argmax(planes[1:], axis=0)
    return 1 - planes[0], cls.astype(np.uint8)

"""Training a U-Net on a prepared dataset and exporting it for the app (V0.15).

Requires the optional dependencies (``pip install 'sedcoretool[unet]'``): PyTorch, segmentation-models-pytorch,
onnx. A consumer GPU (e.g. 8–12 GB) is enough; CPU works for tests and very small sets.

Choices (documented in docs/TECHNICAL_DOCUMENTATION.md):
* U-Net with a ResNet-34 encoder pre-trained on ImageNet (or random weights with ``encoder_weights=None``);
* loss = cross-entropy (binary: BCE with logits) + soft Dice, both ignoring pixels labelled 255;
* augmentation: horizontal and vertical flips and photometric jitter (brightness, contrast, gamma).
  No 90° rotations: trace orientation relative to bedding is diagnostic;
* AdamW, cosine learning-rate schedule, early stopping on the validation Dice;
* export: ONNX (opset 17, input ``image`` 1×3×T×T, output ``logits`` 1×C×T×T) and the JSON read by the app.
"""

from __future__ import annotations

import gc
import json
import random
import warnings
from pathlib import Path
from uuid import uuid4

import numpy as np
from PIL import Image

from .tiling import IGNORE, UnetConfig

_MIN_GPU_BYTES = 256 * 1024**2
_GPU_FAILURES = (
    "out of memory",
    "cuda error: memory allocation",
    "cublas_status_alloc_failed",
    "cudnn_status_alloc_failed",
    "cuda-capable device is busy or unavailable",
    "cuda error: initialization error",
    "cuda driver initialization failed",
    "cuda driver version is insufficient",
    "not currently implemented for the mps device",
    "not implemented for mps",
)


def _torch():
    try:
        import segmentation_models_pytorch as smp
        import torch
    except ImportError as e:  # pragma: no cover - optional dependency
        raise ImportError("install the optional dependencies: pip install 'sedcoretool[unet]'") from e
    return torch, smp


def augment(img: np.ndarray, mask: np.ndarray, rng: random.Random):
    """Flips and photometric jitter on a uint8 tile; returns float image in [0, 1] and the mask."""
    if rng.random() < 0.5:
        img, mask = img[:, ::-1], mask[:, ::-1]
    if rng.random() < 0.5:
        img, mask = img[::-1], mask[::-1]
    x = img.astype(np.float32) / 255.0
    x = np.clip((x - 0.5) * rng.uniform(0.85, 1.15) + 0.5 + rng.uniform(-0.08, 0.08), 0, 1) ** rng.uniform(0.8, 1.25)
    return np.ascontiguousarray(x), np.ascontiguousarray(mask)


def _loader(folder: Path, cfg: UnetConfig, train: bool, batch: int, seed: int):
    torch, _ = _torch()
    files = sorted((folder / "images").glob("*.png"))
    mean, std = np.asarray(cfg.mean, np.float32), np.asarray(cfg.std, np.float32)

    class DS(torch.utils.data.Dataset):
        def __init__(self):
            self.rng = random.Random(seed)

        def __len__(self):
            return len(files)

        def __getitem__(self, i):
            img = np.asarray(Image.open(files[i]).convert("RGB"))
            m = np.asarray(Image.open(folder / "masks" / files[i].name))
            x, m = augment(img, m, self.rng) if train else (img.astype(np.float32) / 255.0, m)
            x = ((x - mean) / std).transpose(2, 0, 1)
            return torch.from_numpy(np.ascontiguousarray(x, np.float32)), torch.from_numpy(m.astype(np.int64))

    return torch.utils.data.DataLoader(DS(), batch_size=batch, shuffle=train, num_workers=0, drop_last=False)


def _loss(logits, y, n_cls):
    torch, _ = _torch()
    F = torch.nn.functional
    valid = (y != IGNORE).float()
    yv = torch.where(y == IGNORE, torch.zeros_like(y), y)
    if n_cls == 1:
        t = (yv > 0).float()
        ce = (F.binary_cross_entropy_with_logits(logits[:, 0], t, reduction="none") * valid).sum() / valid.sum().clamp(min=1)
        p = torch.sigmoid(logits[:, 0]) * valid
        dice = 1 - (2 * (p * t).sum() + 1) / (p.sum() + (t * valid).sum() + 1)
    else:
        ce = F.cross_entropy(logits, y, ignore_index=IGNORE)
        p = torch.softmax(logits, 1) * valid[:, None]
        oh = F.one_hot(yv, n_cls).permute(0, 3, 1, 2).float() * valid[:, None]
        dice = 1 - ((2 * (p * oh).sum((0, 2, 3)) + 1) / (p.sum((0, 2, 3)) + oh.sum((0, 2, 3)) + 1))[1:].mean()
    return ce + dice


def _dice(logits, y, n_cls):
    """Pixel Dice of bioturbation (any class > 0) over valid pixels: (2TP, FP+FN)."""
    pred = (logits[:, 0] > 0) if n_cls == 1 else (logits.argmax(1) > 0)
    valid, t = y != IGNORE, (y > 0) & (y != IGNORE)
    p = pred & valid
    return int((p & t).sum()) * 2, int((p & ~t).sum() + (~p & t).sum())


def train(data: str | Path, out: str | Path, encoder: str = "resnet34", encoder_weights: str | None = "imagenet",
          epochs: int = 60, batch: int = 8, lr: float = 3e-4, patience: int = 12, seed: int = 42,
          device: str | None = None, log=print, progress=None, resume: bool = False, *, cpu_only: bool = False) -> dict:
    """Train on ``data`` (written by ``prepare_dataset``); write ``out.onnx`` and ``out.json``; return the summary.

    ``progress(dict)`` is called after every epoch (epoch, epochs, loss, val_dice, best, best_epoch).
    A checkpoint ``out.ckpt`` is written after every epoch; ``resume=True`` continues from it.
    A CUDA/MPS memory failure retries on CPU. ``cpu_only`` skips GPU probing.
    """
    torch, smp = _torch()
    data, out = Path(data), Path(out).with_suffix("")
    out.parent.mkdir(parents=True, exist_ok=True)
    info = json.loads((data / "dataset.json").read_text(encoding="utf-8"))
    n_cls = 1 if info["binary"] else len(info["classes"]) + 1
    cfg = UnetConfig(tile=info["tile"], overlap=info["overlap"], px_per_cm=info["pxPerCm"], classes=info["classes"])
    min_memory = max(_MIN_GPU_BYTES, batch * cfg.tile**2 * (3 * 4 + 8))
    selected = _select_device(torch, device, min_memory, cpu_only)
    ckpt = out.with_suffix(".ckpt")

    def save_epoch(state, best_path, improved):
        nonlocal resume
        pending = best_path.with_name(f".{best_path.name}.{uuid4().hex}.tmp") if improved else None
        try:
            if pending:
                torch.save(state["model"], pending)
                state["pending_best"] = pending.name
            _save_checkpoint(torch, state, ckpt)
        except BaseException:
            if pending:
                pending.unlink(missing_ok=True)
            raise
        resume = True
        if pending:
            pending.replace(best_path)

    options = (torch, smp, data, out, info, cfg, n_cls, encoder, encoder_weights, epochs, batch, lr, patience, seed, log, progress)

    def attempt(target, *, log_resume=True):
        # Notebook users may have set a global default device. Place implicit allocations on CPU.
        with torch.device("cpu"):
            return _train_on_device(*options, target, resume, save_epoch, log_resume=log_resume)

    try:
        return attempt(selected)
    except RuntimeError as exc:
        if selected.split(":")[0] not in {"cuda", "mps"} or not _is_gpu_failure(torch, exc):
            raise
    # Leave the handler before retrying so the exception traceback releases GPU tensors.
    _release_device(torch, selected)
    return attempt("cpu", log_resume=False)


def _save_checkpoint(torch, state, path):
    tmp = path.with_name(f".{path.name}.{uuid4().hex}.tmp")
    try:
        torch.save(state, tmp)
        tmp.replace(path)
    finally:
        tmp.unlink(missing_ok=True)


def _train_on_device(torch, smp, data, out, info, cfg, n_cls, encoder, encoder_weights, epochs, batch,
                     lr, patience, seed, log, progress, device, resume, save_epoch, *, log_resume):
    random.seed(seed)
    np.random.seed(seed)
    torch.random.default_generator.manual_seed(seed)
    if device.split(":")[0] == "cuda":
        torch.cuda.manual_seed_all(seed)
    elif device.split(":")[0] == "mps" and hasattr(torch.mps, "manual_seed"):
        torch.mps.manual_seed(seed)
    model = smp.Unet(encoder, encoder_weights=encoder_weights, in_channels=3, classes=n_cls).to(device)
    tr = _loader(data / "train", cfg, True, batch, seed)
    has_val = any((data / "val" / "images").glob("*.png"))
    va = _loader(data / ("val" if has_val else "train"), cfg, False, batch, seed)
    opt = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=max(1, epochs))
    best, best_ep, history, wait, start = -1.0, 0, [], 0, 1
    ckpt, best_path = out.with_suffix(".ckpt"), out.with_suffix(".pt")
    if resume and ckpt.exists():
        # Load the checkpoint in host RAM before copying state into the model and optimiser.
        c = torch.load(ckpt, map_location="cpu", weights_only=False)
        pending = c.get("pending_best")
        if pending and pending == Path(pending).name and pending.startswith(f".{best_path.name}."):
            saved_best = best_path.with_name(pending)
            if saved_best.exists():
                saved_best.replace(best_path)
        model.load_state_dict(c["model"]), opt.load_state_dict(c["opt"]), sched.load_state_dict(c["sched"])
        best, best_ep, history, wait, start = c["best"], c["best_ep"], c["history"], c["wait"], c["epoch"] + 1
        if log_resume:
            log(f"resuming after epoch {c['epoch']}")
        del c
    stopped = wait >= patience
    for ep in range(start, epochs + 1):
        if stopped:
            break
        model.train()
        tot = 0.0
        for x, y in tr:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            loss = _loss(model(x), y, n_cls)
            loss.backward()
            opt.step()
            tot += float(loss) * len(x)
        sched.step()
        model.eval()
        a = b = 0
        with torch.no_grad():
            for x, y in va:
                da, db = _dice(model(x.to(device)), y.to(device), n_cls)
                a, b = a + da, b + db
        d = a / (a + b) if a + b else 0.0
        history.append({"epoch": ep, "loss": tot / max(1, len(tr.dataset)), "val_dice": d})
        log(f"epoch {ep}: loss {history[-1]['loss']:.4f}  val Dice {d:.4f}")
        improved = d > best
        if improved:
            best, best_ep, wait = d, ep, 0
        else:
            wait += 1
            stopped = wait >= patience
        save_epoch({"model": model.state_dict(), "opt": opt.state_dict(), "sched": sched.state_dict(), "epoch": ep,
                    "best": best, "best_ep": best_ep, "history": history, "wait": wait}, best_path, improved)
        if progress:
            progress({"epoch": ep, "epochs": epochs, "loss": history[-1]["loss"], "val_dice": d, "best": best,
                      "best_epoch": best_ep, "device": device})
    model = model.cpu()
    model.load_state_dict(torch.load(best_path, map_location="cpu"))
    export_onnx(model.eval(), cfg.tile, out.with_suffix(".onnx"))
    summary = {**cfg.to_json(), "encoder": encoder, "encoderWeights": encoder_weights, "bestEpoch": best_ep,
               "valDice": best, "validatedOn": "val" if has_val else "train (no validation cores)",
               "split": info["split"], "seed": seed, "history": history}
    out.with_suffix(".json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
    return summary


def _select_device(torch, device, min_memory: int, cpu_only: bool) -> str:
    if cpu_only:
        return "cpu"
    requested = str(device) if device else None
    kind = requested.split(":")[0] if requested is not None else None
    if requested is not None and kind not in {"cuda", "mps"}:
        return requested
    try:
        if kind in {None, "cuda"}:
            with warnings.catch_warnings():
                warnings.filterwarnings("ignore", message=r"CUDA initialization:.*", category=UserWarning)
                if torch.cuda.is_available():
                    target = requested or "cuda"
                    free, _ = torch.cuda.mem_get_info(target)
                    return target if free >= min_memory else "cpu"
        mps = getattr(torch.backends, "mps", None)
        if kind in {None, "mps"} and mps is not None and mps.is_available():
            free = torch.mps.recommended_max_memory() - torch.mps.driver_allocated_memory()
            return (requested or "mps") if free >= min_memory else "cpu"
    except (RuntimeError, OSError, AttributeError, AssertionError):
        # Older runtimes may expose the accelerator without a working memory probe.
        pass
    return "cpu"


def _is_gpu_failure(torch, exc: RuntimeError) -> bool:
    oom = getattr(torch, "OutOfMemoryError", getattr(torch.cuda, "OutOfMemoryError", None))
    message = str(exc).lower()
    if "cpuallocator" in message:
        return False
    return (oom is not None and isinstance(exc, oom)) or any(fragment in message for fragment in _GPU_FAILURES)


def _release_device(torch, device: str) -> None:
    gc.collect()
    try:
        if device.split(":")[0] == "cuda":
            torch.cuda.empty_cache()
        else:
            torch.mps.empty_cache()
    except (RuntimeError, OSError, AttributeError, AssertionError):
        pass


def best_device(*, cpu_only: bool = False) -> str:
    """CUDA or MPS with verified headroom, or CPU."""
    torch, _ = _torch()
    return _select_device(torch, None, _MIN_GPU_BYTES, cpu_only)


def device_info(*, cpu_only: bool = False) -> dict:
    """Training hardware seen by PyTorch (for the app's training dialog)."""
    torch, _ = _torch()
    d = _select_device(torch, None, _MIN_GPU_BYTES, cpu_only)
    try:
        name = torch.cuda.get_device_name() if d == "cuda" else ("Apple GPU (MPS)" if d == "mps" else "CPU")
    except (RuntimeError, OSError, AttributeError, AssertionError):
        d, name = "cpu", "CPU"
    return {"device": d, "name": name, "torch": torch.__version__}


def export_onnx(model, tile: int, path: str | Path) -> Path:
    torch, _ = _torch()
    x = torch.zeros(1, 3, tile, tile, device="cpu")
    kw = dict(input_names=["image"], output_names=["logits"], opset_version=17,
              dynamic_axes={"image": {0: "batch"}, "logits": {0: "batch"}})
    try:
        torch.onnx.export(model, x, str(path), dynamo=False, **kw)
    except TypeError:  # PyTorch < 2.5 has no `dynamo` argument
        torch.onnx.export(model, x, str(path), **kw)
    return Path(path)

## SedCoreTool V0.123

SedCoreTool quantifies bioturbation and ichnotaxa in sediment-core images and combines them with other down-core
records and described intervals. Version 0.x, in active development. The downloads contain no sample data: the
program starts empty and you open your own images and tables.

### Highlights
- New name: **SedCoreTool** (formerly CorAnTool and IchAnTool). Projects saved with earlier versions and `.corantool`
  packages still open.
- Editable colours in every figure, including the multiproxy and location maps; invalid stored colours fall back to
  the defaults.
- Final test on reserved cores excludes only the training examples that may come from them.
- Age models tolerate repeated rows and incomplete ranges.
- If saving fails on closing, the program asks whether to retry, export the project or close without saving.
- Python package: exclusion layer, batch analysis of many sections and cores from a CSV manifest, LAS 2.0 export.

The full list of changes is in [CHANGELOG.md](https://github.com/SantiCa/SedCoreTool/blob/master/CHANGELOG.md).

### Downloads

| System | File |
|---|---|
| Windows 10/11, x64 | `SedCoreTool-V0.123-windows-setup.exe` or `SedCoreTool-V0.123-windows-portable.exe` |
| macOS 13 or later, Intel and Apple Silicon | `SedCoreTool-V0.123-mac-universal.dmg` |
| Linux, x64 | `SedCoreTool-V0.123-linux-x86_64.AppImage` or `SedCoreTool-V0.123-linux-amd64.deb` |
| Any system, in a browser | `SedCoreTool_V0.123.html` (single file, works offline) |
| Verification | `SHA256SUMS.txt` and `RELEASE_MANIFEST.json` |

Windows: run the installer or open the portable version. macOS: drag the application to Applications. Linux: make
the AppImage executable or install the DEB package. The HTML file opens with a double click. The installers are not
code-signed, so the operating system may ask for confirmation the first time.

To verify a file: `Get-FileHash <file> -Algorithm SHA256` in PowerShell, `shasum -a 256 <file>` on macOS or
`sha256sum -c SHA256SUMS.txt` on Linux. The manifest identifies the source commit and the size and hash of each file.

### Before you update
Export a copy of your projects. The program does not update itself; install the new version over the old one.

### Known limitations
- U-Net models must be trained by the user; no model trained on real cores is distributed yet.
- The first download of the vision model and map backgrounds need an internet connection.
- Training a U-Net needs Python 3.10 or later on the computer and installs a 2–3 GB module once.

Documentation: [technical documentation](https://github.com/SantiCa/SedCoreTool/blob/master/docs/TECHNICAL_DOCUMENTATION.md).

// V0.17: training of U-Net models on the user's computer, from the desktop program.
// The app sends the training ZIP (same as «U-Net training data»); this module runs the Python package
// (`python -m sedcoretool.cli unet prepare|train`) in a private virtual environment and reports progress.
// The environment (PyTorch + sedcoretool[unet], 2–3 GB) is installed on demand with the system Python ≥ 3.10.
// Plain Node (no Electron imports), so it can be tested on its own.
const fs = require('node:fs');
const path = require('node:path');
const os = require('node:os');
const { spawn, spawnSync } = require('node:child_process');

const WIN = process.platform === 'win32';
const safe = (s) => String(s || 'model').replace(/[^\w.-]+/g, '_').slice(0, 60) || 'model';

class Trainer {
  /**
   * root: folder for the environment, runs and models (Electron: userData/trainer).
   * pySrc: folder with the Python package source (pyproject.toml + src/).
   * send(event, data): progress events to the window ('log', 'progress', 'done', 'error', 'state').
   * python: optional interpreter that already has sedcoretool[unet] (skips the private environment; for tests).
   */
  constructor({ root, pySrc, send = () => {}, python = process.env.SEDCORETOOL_PYTHON || process.env.CORANTOOL_PYTHON || null }) {
    Object.assign(this, { root, pySrc, send, fixedPython: python, child: null, busy: null });
    fs.mkdirSync(path.join(root, 'models'), { recursive: true });
    fs.mkdirSync(path.join(root, 'runs'), { recursive: true });
  }

  get venv() { return path.join(this.root, 'venv'); }
  get venvPython() { return path.join(this.venv, WIN ? 'Scripts/python.exe' : 'bin/python'); }
  python() { return this.fixedPython || (fs.existsSync(this.venvPython) ? this.venvPython : null); }

  /** System Python ≥ 3.10 used to create the environment: { cmd, args, version } or null. */
  systemPython() {
    const cands = WIN ? [['py', ['-3']], ['python', []], ['python3', []]] : [['python3', []], ['python', []]];
    for (const [cmd, pre] of cands) {
      const r = spawnSync(cmd, [...pre, '-c', 'import sys;print("%d.%d" % sys.version_info[:2])'], { encoding: 'utf8', timeout: 15000 });
      const v = (r.stdout || '').trim(), [a, b] = v.split('.').map(Number);
      if (r.status === 0 && (a > 3 || (a === 3 && b >= 10))) return { cmd, args: pre, version: v };
    }
    return null;
  }

  hasNvidia() { return spawnSync('nvidia-smi', ['-L'], { encoding: 'utf8', timeout: 15000 }).status === 0; }

  /** Use the bundled application code with the dependencies from the managed interpreter. */
  pythonEnv(bundled = true) {
    const src = this.pySrc && path.join(this.pySrc, 'src');
    const env = { ...process.env, PYTHONUNBUFFERED: '1', PYTHONIOENCODING: 'utf-8' };
    if (bundled && !this.fixedPython && src && fs.existsSync(path.join(src, 'sedcoretool', 'cli.py'))) {
      env.PYTHONPATH = [src, env.PYTHONPATH].filter(Boolean).join(path.delimiter);
    }
    return env;
  }

  /** What the dialog shows: interpreter, whether the module is ready, training hardware, models and an unfinished run. */
  async status() {
    const py = this.python(), out = { python: py, installed: false, device: null, system: null, models: this.models(), busy: this.busy, platform: process.platform };
    if (!py) { out.system = this.systemPython(); return out; }
    const r = spawnSync(py, ['-m', 'sedcoretool.cli', 'unet', 'device'], { encoding: 'utf8', timeout: 120000, env: this.pythonEnv() });
    if (r.status === 0) { out.installed = true; try { out.device = JSON.parse(r.stdout.trim().split('\n').pop()); } catch { /* keep null */ } }
    else out.error = (r.stderr || '').trim().split('\n').slice(-3).join('\n');
    return out;
  }

  /** Run a command, sending each output line; resolves with the exit code. */
  run(cmd, args, opt = {}) {
    return new Promise((resolve, reject) => {
      const cli = args[0] === '-m' && args[1] === 'sedcoretool.cli';
      const env = this.pythonEnv(cli);
      const p = spawn(cmd, args, { cwd: opt.cwd || this.root, env, windowsHide: true });
      this.child = p;
      const feed = (buf) => {
        for (const line of String(buf).split(/\r?\n/)) {
          if (!line.trim()) continue;
          if (line.startsWith('{"progress"')) { try { this.send('progress', JSON.parse(line).progress); continue; } catch { /* plain line */ } }
          this.send('log', line);
        }
      };
      p.stdout.on('data', feed); p.stderr.on('data', feed);
      p.on('error', reject);
      p.on('close', (code, signal) => { this.child = null; resolve(signal ? -1 : code); });
    });
  }

  async step(label, cmd, args) {
    this.send('log', `▸ ${label}`);
    const code = await this.run(cmd, args);
    if (code !== 0) throw new Error(code === -1 ? 'cancelled' : `${label}: exit code ${code}`);
  }

  /** Private environment: venv + PyTorch (CUDA build on Windows with an NVIDIA card) + sedcoretool[unet] from pySrc. */
  async install() {
    if (this.busy) throw new Error('busy');
    const sys = this.systemPython();
    if (!sys) throw new Error('python-missing');
    this.busy = 'install'; this.send('state', { busy: this.busy });
    try {
      if (!fs.existsSync(this.venvPython)) await this.step(`Python ${sys.version}: virtual environment`, sys.cmd, [...sys.args, '-m', 'venv', this.venv]);
      const py = this.venvPython;
      await this.step('pip', py, ['-m', 'pip', 'install', '--upgrade', 'pip']);
      const torch = ['-m', 'pip', 'install', 'torch'];
      if (WIN && this.hasNvidia()) torch.push('--index-url', 'https://download.pytorch.org/whl/cu128');   // PyPI wheels for Windows are CPU-only
      await this.step('PyTorch', py, torch);
      const src = path.join(os.tmpdir(), `sedcoretool-py-${Date.now()}`);   // pip builds in the source folder: use a writable copy
      fs.cpSync(this.pySrc, src, { recursive: true });
      try { await this.step('sedcoretool[unet]', py, ['-m', 'pip', 'install', `${src}[unet]`]); } finally { fs.rmSync(src, { recursive: true, force: true }); }
      this.send('log', '✓');
    } finally { this.busy = null; this.send('state', { busy: null }); }
  }

  /**
   * Train from the app's ZIP. opt = { zip (Uint8Array), name, val: [], test: [], epochs, pretrained, tile, overlap,
   * encoder, resume }. Resume reuses the prepared tiles and the checkpoint of the same name.
   */
  async train(opt) {
    if (this.busy) throw new Error('busy');
    const py = this.python(); if (!py) throw new Error('not-installed');
    const name = safe(opt.name), run = path.join(this.root, 'runs', name), data = path.join(run, 'data');
    const model = path.join(this.root, 'models', name);
    this.busy = 'train'; this.send('state', { busy: this.busy, name });
    try {
      const resume = !!opt.resume && fs.existsSync(path.join(data, 'dataset.json')) && fs.existsSync(model + '.ckpt');
      if (!resume) {
        fs.rmSync(run, { recursive: true, force: true }); fs.mkdirSync(run, { recursive: true });
        for (const ext of ['.ckpt', '.pt', '.onnx', '.json']) fs.rmSync(model + ext, { force: true });
        fs.writeFileSync(path.join(run, 'dataset.zip'), Buffer.from(opt.zip));
        const a = ['-m', 'sedcoretool.cli', 'unet', 'prepare', path.join(run, 'dataset.zip'), data, '--tile', String(opt.tile || 512), '--overlap', String(opt.overlap || 64)];
        if (opt.val?.length) a.push('--val', ...opt.val);
        if (opt.test?.length) a.push('--test', ...opt.test);
        await this.step('prepare', py, a);
      }
      const a = ['-m', 'sedcoretool.cli', 'unet', 'train', data, model, '--epochs', String(opt.epochs || 60), '--encoder', opt.encoder || 'resnet34', '--progress-json'];
      if (opt.pretrained === false) a.push('--no-pretrained');
      if (resume) a.push('--resume');
      await this.step(resume ? 'train (resume)' : 'train', py, a);
      const ev = path.join(run, 'evaluation_test.json');
      const test = JSON.parse(fs.readFileSync(path.join(data, 'dataset.json'), 'utf8')).split.test.cores;
      if (test.length) await this.step('evaluate (test cores)', py, ['-m', 'sedcoretool.cli', 'unet', 'evaluate', path.join(run, 'dataset.zip'), model + '.onnx', '--dataset', data, '-o', ev]);
      this.send('done', { name, evaluation: fs.existsSync(ev) ? JSON.parse(fs.readFileSync(ev, 'utf8')) : null });
      return name;
    } finally { this.busy = null; this.send('state', { busy: null }); }
  }

  cancel() { if (this.child) this.child.kill(); }

  /** Trained models: name, validation Dice, best epoch, date and test score if evaluated. */
  models() {
    const dir = path.join(this.root, 'models');
    return fs.readdirSync(dir).filter((f) => f.endsWith('.json') && fs.existsSync(path.join(dir, f.replace(/\.json$/, '.onnx')))).map((f) => {
      const name = f.replace(/\.json$/, ''), j = JSON.parse(fs.readFileSync(path.join(dir, f), 'utf8'));
      const ev = path.join(this.root, 'runs', name, 'evaluation_test.json');
      const test = fs.existsSync(ev) ? JSON.parse(fs.readFileSync(ev, 'utf8')).pooled : null;
      return { name, valDice: j.valDice, bestEpoch: j.bestEpoch, split: j.split, date: fs.statSync(path.join(dir, f)).mtime.toISOString(), test };
    }).sort((a, b) => (a.date < b.date ? 1 : -1));
  }

  /** Files of a trained model, for loading it in the app. */
  readModel(name) {
    const base = path.join(this.root, 'models', safe(name));
    return { name: safe(name), onnx: fs.readFileSync(base + '.onnx'), json: fs.readFileSync(base + '.json', 'utf8') };
  }

  /** A model whose last run stopped before the end (can be resumed). */
  unfinished(name) { const b = path.join(this.root, 'models', safe(name)); return fs.existsSync(b + '.ckpt') && !fs.existsSync(b + '.onnx'); }
}

module.exports = { Trainer, safe };

def rle(mask):
    flat = mask.ravel()
    edges = np.r_[0, np.flatnonzero(flat[1:] != flat[:-1]) + 1, flat.size]
    return np.column_stack((flat[edges[:-1]], np.diff(edges))).ravel().tolist()


def make_synthetic(path, *, height=6400):
    """Write a deterministic normal project package with precomputed nonoverlapping traces."""
    if not isinstance(height, int) or not 256 <= height <= 16000:
        raise ValueError("Synthetic section height must be an integer from 256 to 16000 pixels")
    rng = np.random.Generator(np.random.PCG64(SEED))
    w, h, depth_cm = 960, height, 128
    filters = dict(channel="rgb", blackPoint=0, whitePoint=255, gamma=1, brightness=0, contrast=0, saturation=100)
    classes = [
        dict(id="undet", name="Undetermined", color="#d9a441", counts=True, fixed=True, plain=True),
        dict(id="chondrites", name="Chondrites", color="#4c8dff", counts=True),
        dict(id="planolites", name="Planolites", color="#ef5b3b", counts=True),
        dict(id="artifact", name="Artifact", color="#8c8c8c", counts=False, fixed=True, plain=True),
    ]
    sections = []
    with ZipFile(path, "w") as archive:
        for section in range(8):
            name = f"synthetic-{section + 1:02d}"
            mask = np.zeros((h, w), dtype=np.uint16)
            pixels = rng.integers(165, 201, size=(h, w), dtype=np.uint8)
            objects = []
            for y0 in range(16, h - 100, 128):
                for x0, max_width, cls in [(32, 430, "chondrites"), (480, 430, "planolites")]:
                    x1 = x0 + int(rng.integers(40, max_width))
                    y1 = y0 + int(rng.integers(28, 101))
                    ident = len(objects) + 1
                    mask[y0:y1, x0:x1] = ident
                    shade = int(rng.integers(65, 136))
                    pixels[y0:y1, x0:x1] = shade
                    xx, yy = np.arange(x0, x1), np.arange(y0, y1)
                    dx, dy = len(xx), len(yy)
                    n = dx * dy
                    stats = dict(n=n, sx=int(xx.sum()) * dy, sy=int(yy.sum()) * dx,
                                 sxx=int((xx**2).sum()) * dy, syy=int((yy**2).sum()) * dx,
                                 sxy=int(xx.sum()) * int(yy.sum()), x0=x0, x1=x1 - 1, y0=y0, y1=y1 - 1,
                                 per=2 * (dx + dy) - 4, sL=n * shade, sLL=n * shade**2, sA=0, sB=0, rL=0, rN=0)
                    objects.append(dict(id=ident, cls=cls, src="draw", st=stats))
            excluded = np.zeros((h, w), dtype=np.uint8)
            excluded[2000:2125] = 1
            excluded[3100:3500, 260:520] = 1
            image_path = f"images/{name}.jpg"
            image = io.BytesIO()
            Image.fromarray(pixels).save(image, format="JPEG", quality=85)
            archive.writestr(ZipInfo(image_path), image.getvalue())
            sections.append(dict(
                id=name, name=name, w=w, h=h, full=dict(w=w, h=h), ds=1, rot=0,
                file=dict(path=image_path, name=f"{name}.jpg", type="image/jpeg"),
                top=section * depth_cm, bottom=(section + 1) * depth_cm, units="cm",
                roi=dict(x0=20, x1=w - 20, y0=0, y1=h), filters=filters, preset="none",
                detect=dict(method="adaptive", polarity="dark"), nextId=len(objects) + 1,
                labels=rle(mask), exclude=rle(excluded), objects=objects,
            ))
        project = dict(app="SedCoreTool", kind="project", version=5, id="benchmark-synthetic", name="Benchmark synthetic",
                       classes=classes, settings=dict(binCm=2, minValid=20, noDataPad=0),
                       cores=[dict(id="synthetic", name="Synthetic", units="cm", sections=sections)])
        archive.writestr(ZipInfo("project.json"), json.dumps(project, separators=(",", ":")))


def unpack(runs, section):
    pairs = np.asarray(runs, dtype=np.int32).reshape(-1, 2)
    height, width = section["h"], section["w"]
    if np.any(pairs[:, 1] <= 0) or pairs[:, 1].sum() != height * width:
        raise ValueError(f"Invalid mask length in {section['name']}")
    roi = section["roi"]
    return np.repeat(pairs[:, 0], pairs[:, 1]).reshape(height, width)[roi["y0"]:roi["y1"], roi["x0"]:roi["x1"]]


def python_trial(snapshot_file):
    snapshot = json.loads(snapshot_file.read_text(encoding="utf-8"))
    classes, settings = snapshot["classes"], snapshot["settings"]
    inputs = []
    for section in snapshot["sections"]:
        ids = unpack(section["labels"], section)
        if ids.min() < 0:
            raise ValueError("These benchmark fixtures must have nonoverlapping labels")
        lookup = np.asarray(section["eff"], dtype=np.int32) + 1
        exclude = unpack(section["exclude"], section).astype(bool)
        inputs.append((lookup[ids], exclude, section))
    start = perf_counter()
    results = [
        analyze_mask(mask, section["top"], section["bottom"], units=section["units"],
                     bin_cm=settings["binCm"], min_valid=settings["minValid"], keep_partial=True,
                     class_names=classes, exclude=exclude, name=section["name"])
        for mask, exclude, section in inputs
    ]
    elapsed = (perf_counter() - start) * 1000
    columns = ["depth_top", "depth_bottom", "depth", "valid_pct", "bioturbation_pct",
               *(f"{cls}_pct" for cls in classes), "BI", "ii"]
    bins = [json.loads(result.data[columns].to_json(orient="values", double_precision=15)) for result in results]
    print(json.dumps({"analysisMs": elapsed, "bins": bins, "workerPid": os.getpid()}))


def verify_bins(browser, python):
    if len(browser) != len(python):
        raise ValueError("Section count differs between JavaScript and Python")
    max_depth, max_pct = 0.0, 0.0
    for section, (js_rows, py_rows) in enumerate(zip(browser, python, strict=True), start=1):
        if len(js_rows) != len(py_rows):
            raise ValueError(f"Interval count differs in section {section}")
        for row, (js, py) in enumerate(zip(js_rows, py_rows, strict=True), start=1):
            if len(js) != len(py) or js[-2:] != py[-2:]:
                raise ValueError(f"Class columns or BI/ii differ in section {section}, interval {row}")
            for col, (a, b) in enumerate(zip(js[:-2], py[:-2], strict=True)):
                if a is None or b is None:
                    if a is not b:
                        raise ValueError(f"Missing value differs in section {section}, interval {row}, column {col + 1}")
                    continue
                error = abs(a - b)
                if not error <= ABS_TOL:
                    raise ValueError(f"Numerical mismatch in section {section}, interval {row}, column {col + 1}: {a} vs {b}")
                if col < 3:
                    max_depth = max(max_depth, error)
                else:
                    max_pct = max(max_pct, error)
    return {"max_depth_error": max_depth, "max_percentage_point_error": max_pct}


def run_sample(command, *, browser_tree=False):
    peaks, samples = Counter(), Counter()
    with tempfile.TemporaryFile() as output, subprocess.Popen(command, cwd=ROOT, stdout=output) as child:
        root = psutil.Process(child.pid)
        start = perf_counter()
        try:
            while child.poll() is None:
                resident = {}
                try:
                    descendants = root.children(recursive=True)
                    processes = descendants if browser_tree else [root, *descendants]
                    for process in processes:
                        try:
                            rss = process.memory_info().rss
                            if rss:
                                resident[process.pid] = rss
                        except psutil.NoSuchProcess:
                            pass
                except psutil.NoSuchProcess:
                    pass
                if browser_tree and resident:
                    resident = {0: sum(resident.values())}
                for pid, rss in resident.items():
                    peaks[pid] = max(peaks[pid], rss)
                    samples[pid] += 1
                try:
                    child.wait(timeout=SAMPLE_SECONDS)
                except subprocess.TimeoutExpired:
                    if perf_counter() - start >= TIMEOUT:
                        raise TimeoutError(f"Benchmark process exceeded {TIMEOUT} seconds") from None
        finally:
            if child.poll() is None:
                for process in reversed(root.children(recursive=True)):
                    try:
                        process.kill()
                    except psutil.NoSuchProcess:
                        pass
                child.kill()
        if child.returncode:
            raise subprocess.CalledProcessError(child.returncode, command)
        output.seek(0)
        result = json.load(output)
        # A Windows virtualenv can launch the real Python worker as a child of its executable shim.
        pid = 0 if browser_tree else result.pop("workerPid")
        if not samples[pid]:
            raise RuntimeError("No RSS samples were captured for the benchmark worker")
        return result, peaks[pid] / MIB, samples[pid]


def digest(data):
    return hashlib.sha256(data).hexdigest()


def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT)


def source_metadata(extra_files=()):
    files = set(git("ls-files", "-z", "--cached", "--others", "--exclude-standard", "--", "app", "src/sedcoretool", "examples/U1385I-1H").decode().split("\0")) - {""}
    files.update(["tools/benchmark.py", "tools/benchmark_browser.mjs"])
    files.update(extra_files)
    hashes = hashlib.sha256()
    for name in sorted(files):
        data = (ROOT / name).read_bytes()
        if Path(name).suffix in {".css", ".csv", ".html", ".js", ".json", ".md", ".mjs", ".py", ".svg", ".txt"}:
            data = data.replace(b"\r\n", b"\n")
        hashes.update(name.encode() + b"\0" + hashlib.sha256(data).digest())
    return {
        "commit": git("rev-parse", "HEAD").decode().strip(),
        "dirty": bool(git("status", "--porcelain")),
        "input_files_sha256": hashes.hexdigest(),
    }


def describe_environment():
    cpu = platform.processor()
    if sys.platform == "win32":
        import winreg

        with winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, r"HARDWARE\DESCRIPTION\System\CentralProcessor\0") as key:
            cpu = winreg.QueryValueEx(key, "ProcessorNameString")[0].strip()
    elif sys.platform == "linux":
        for line in Path("/proc/cpuinfo").read_text().splitlines():
            if line.startswith("model name"):
                cpu = line.split(":", 1)[1].strip()
                break
    elif sys.platform == "darwin":
        cpu = subprocess.check_output(["sysctl", "-n", "machdep.cpu.brand_string"], text=True).strip()
    return {
        "os": platform.platform(), "architecture": platform.machine(), "cpu": cpu,
        "logical_cpus": psutil.cpu_count(), "physical_cores": psutil.cpu_count(logical=False),
        "ram_mib": round(psutil.virtual_memory().total / MIB),
        "python": platform.python_version(), "sedcoretool": sedcoretool.__version__,
        "numpy": np.__version__, "pandas": pd.__version__, "pillow": pillow_version, "psutil": psutil.__version__,
    }


class QuietHandler(SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, directory=str(ROOT), **kwargs)

    def log_message(self, *_):
        pass


def browser_args(action, url, profile, chrome, case, extra):
    return ["node", str(DRIVER), action, url, str(profile), chrome, case, str(extra)]


def benchmark_case(case, args, url, workspace):
    case_dir = workspace / case
    case_dir.mkdir()
    profile, fixture = case_dir / "seed", case_dir / "synthetic.sedcoretool"
    print(f"Preparing {case}", file=sys.stderr, flush=True)
    if case == "synthetic":
        make_synthetic(fixture)
    prepare = browser_args("prepare", url, profile, args.chrome, case, fixture)
    runtimes = json.loads(subprocess.check_output(prepare, cwd=ROOT, timeout=TIMEOUT))
    snapshot_file = case_dir / "analysis-input.json"
    runs, reference_bins, reference_edit, snapshot = [], None, None, None
    for i in range(args.warmups + args.repetitions):
        with tempfile.TemporaryDirectory(prefix="trial-", dir=case_dir) as trial_dir:
            trial_profile = Path(trial_dir) / "profile"
            shutil.copytree(profile, trial_profile)
            browser, browser_peak, browser_samples = run_sample(
                browser_args("measure", url, trial_profile, args.chrome, case, "snapshot" if i == 0 else "verify"),
                browser_tree=True,
            )
        edit = {key: browser[key] for key in ("id", "bbox", "count")}
        if i == 0:
            snapshot = browser.pop("snapshot")
            reference_bins = browser["bins"]
            reference_edit = edit
            snapshot_file.write_text(json.dumps(snapshot, separators=(",", ":")), encoding="utf-8")
        elif edit != reference_edit or browser["bins"] != reference_bins:
            raise ValueError(f"{case} did not reproduce its browser edit and interval results on repetition {i + 1}")
        python, python_peak, python_samples = run_sample([sys.executable, __file__, "--python-worker", str(snapshot_file)])
        parity = verify_bins(browser["bins"], python["bins"])
        entry = {
            "kind": "warmup" if i < args.warmups else "measurement", "open_ms": browser["openMs"],
            "autosave_ms": browser["saveMs"], "javascript_analysis_ms": browser["analysisMs"],
            "python_analysis_ms": python["analysisMs"], "browser_peak_rss_mib": browser_peak,
            "python_peak_rss_mib": python_peak, "browser_rss_samples": browser_samples, "python_rss_samples": python_samples,
            "parity": parity,
        }
        runs.append(entry)
        print(f"{case} {entry['kind']} {i + 1}: open {entry['open_ms']:.1f} ms, save {entry['autosave_ms']:.1f} ms, "
              f"JS {entry['javascript_analysis_ms']:.1f} ms, Python {entry['python_analysis_ms']:.1f} ms", file=sys.stderr)
    measurements = [run for run in runs if run["kind"] == "measurement"]
    summary = {}
    for metric in ("open_ms", "autosave_ms", "javascript_analysis_ms", "python_analysis_ms",
                   "browser_peak_rss_mib", "python_peak_rss_mib"):
        values = [run[metric] for run in measurements]
        summary[metric] = {"median": statistics.median(values), "min": min(values), "max": max(values)}
    sections = [{k: v for k, v in section.items() if k not in {"labels", "exclude", "stacks", "eff"}}
                for section in snapshot["sections"]]
    return {
        "case": case, "runtimes": runtimes, "seed": SEED if case == "synthetic" else None,
        "synthetic_package_sha256": digest(fixture.read_bytes()) if case == "synthetic" else None,
        "analysis_input_sha256": digest(snapshot_file.read_bytes()),
        "browser_intervals_sha256": digest(json.dumps(reference_bins, separators=(",", ":")).encode()),
        "classes": snapshot["classes"], "settings": snapshot["settings"], "sections": sections,
        "pixels": sum(s["w"] * s["h"] for s in sections),
        "intervals": sum(map(len, reference_bins)),
        "no_data_intervals": sum(row[-2:] == [None, None] for rows in reference_bins for row in rows),
        "deleted_trace": reference_edit,
        "reload_verified_runs": len(runs), "runs": runs, "summary": summary,
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--chrome", default=os.getenv("CHROME"), help="Chrome or Chromium executable (or set CHROME)")
    parser.add_argument("--case", choices=("all", *CASES), default="all")
    parser.add_argument("--warmups", type=int, default=1)
    parser.add_argument("--repetitions", type=int, default=5)
    parser.add_argument("--output", type=Path, help="Write the JSON report here instead of stdout")
    args = parser.parse_args()
    if min(args.warmups, args.repetitions) < 1:
        parser.error("warmups and repetitions must be positive")
    args.chrome = args.chrome or next((path for name in ("google-chrome", "chromium", "chrome") if (path := shutil.which(name))), None)
    if not args.chrome:
        parser.error("Set --chrome to a Chrome or Chromium executable")
    args.chrome = str(Path(args.chrome).resolve(strict=True))
    if Path(sedcoretool.__file__).resolve().parent != ROOT / "src" / "sedcoretool":
        parser.error('Install this checkout with python -m pip install -e "."')
    report = {
        "started_utc": datetime.now(timezone.utc).isoformat(), "source": source_metadata(),
        "environment": describe_environment(), "warmups": args.warmups, "repetitions": args.repetitions,
        "absolute_tolerance": ABS_TOL, "rss_poll_target_ms": SAMPLE_SECONDS * 1000,
    }
    with tempfile.TemporaryDirectory(prefix="sedcoretool-benchmark-") as tmp:
        with ThreadingHTTPServer(("127.0.0.1", 0), QuietHandler) as server:
            server_thread = threading.Thread(target=server.serve_forever, daemon=True)
            server_thread.start()
            try:
                url = f"http://127.0.0.1:{server.server_port}/app/index.html"
                cases = CASES if args.case == "all" else (args.case,)
                report["cases"] = [benchmark_case(case, args, url, Path(tmp)) for case in cases]
            finally:
                server.shutdown()
                server_thread.join()
    output = json.dumps(report, indent=2, allow_nan=False) + "\n"
    if args.output:
        args.output.write_text(output, encoding="utf-8")
    else:
        print(output, end="")


if __name__ == "__main__":
    if len(sys.argv) == 3 and sys.argv[1] == "--python-worker":
        python_trial(Path(sys.argv[2]))
    else:
        main()

// Measure the application's analysis without changing its implementation or persistence paths.
import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import { createRequire } from 'node:module';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

export function summarizeCPU(profile) {
  const nodes = new Map(profile.nodes.map((node) => [node.id, node]));
  const weights = new Map();
  assert.equal(profile.samples.length, profile.timeDeltas.length, 'CPU samples need matching weights');
  for (let index = 0; index < profile.samples.length; index++) {
    const id = profile.samples[index], weight = profile.timeDeltas[index];
    assert.ok(nodes.has(id) && Number.isFinite(weight) && weight >= 0, 'Invalid CPU sample');
    weights.set(id, (weights.get(id) || 0) + weight);
  }
  const sites = [...weights].map(([id, weight]) => {
    const frame = nodes.get(id).callFrame;
    return { function: frame.functionName || '(anonymous)',
      source: frame.url.startsWith('pptr:') ? '(benchmark)' : frame.url.split('/').slice(-2).join('/') || '(runtime)',
      line: frame.lineNumber + 1, sampled_self_ms: weight / 1000 };
  }).sort((a, b) => b.sampled_self_ms - a.sampled_self_ms);
  return { samples: profile.samples.length, sampled_ms: profile.timeDeltas.reduce((sum, dt) => sum + dt, 0) / 1000,
    sites };
}

async function main() {
  const [url, userDataDir, executablePath, snapshotFile, mode] = process.argv.slice(2);
  assert.ok(['measure', 'profile'].includes(mode));
  const expected = JSON.parse(await readFile(snapshotFile, 'utf8'));
  const { default: puppeteer } = await import('puppeteer-core');
  const browser = await puppeteer.launch({ executablePath, userDataDir, headless: true,
    args: ['--no-sandbox'], defaultViewport: { width: 1440, height: 900 } });
  const errors = [];
  try {
    const [page] = await browser.pages();
    page.setDefaultTimeout(120000);
    page.on('pageerror', (error) => errors.push(error.message));
    page.on('console', (message) => {
      if (message.type() === 'error' && !/Failed to load resource/.test(message.text())) errors.push(message.text());
    });
    await page.goto(url, { waitUntil: 'networkidle0' });
    await page.waitForFunction(() => {
      const S = window.ichantool?.S;
      return S?.sections[0]?.enh && document.getElementById('busy').hidden
        && document.getElementById('saveState').dataset.state === 'saved';
    });
    assert.deepEqual(await snapshot(page), expected, 'Edited fixture differs before analysis');
    const scenarios = [];
    const client = mode === 'profile' ? await page.createCDPSession() : null;
    if (client) { await client.send('Profiler.enable'); await client.send('Profiler.setSamplingInterval', { interval: 1000 }); }
    for (const binCm of [2, 1.37]) {
      const scenario = await measure(page, binCm);
      if (client) {
        scenario.cpu = {};
        for (const [cache, passes] of [['cold', 20], ['cached', 200]]) {
          await client.send('Profiler.start');
          await page.evaluate(({ cache, passes }) => {
            const { S, analyze } = window.ichantool;
            for (let pass = 0; pass < passes; pass++) for (const section of S.sections) {
              if (cache === 'cold') delete section._rc;
              analyze(section);
            }
          }, { cache, passes });
          scenario.cpu[cache] = { passes, ...summarizeCPU((await client.send('Profiler.stop')).profile) };
          assert.deepEqual(await bins(page), scenario.bins, 'Profiling changed interval results');
        }
      }
      scenarios.push(scenario);
    }
    if (client) { await client.send('Profiler.disable'); await client.detach(); }
    await page.evaluate(() => { window.ichantool.S.settings.binCm = 2; });
    assert.deepEqual(await snapshot(page), expected, 'Analysis changed input masks, classes or settings');
    assert.deepEqual(errors, [], 'Application errors');
    console.log(JSON.stringify({ scenarios, inputs_unchanged: true,
      chrome: await browser.version(), node: process.version,
      puppeteer: createRequire(import.meta.url)('puppeteer-core/package.json').version }));
  } finally { await browser.close(); }
}

async function snapshot(page) {
  return page.evaluate(async () => {
    const { S, serialize } = window.ichantool, { effClassArray } = await import('./js/app.js');
    const sections = S.sections;
    // Force resident masks before timing, as in M0.1. Never raise the application's memory budget.
    for (const s of sections) {
      if (s.labels.length !== s.w * s.h || s.exclude.length !== s.w * s.h) throw new Error('Invalid masks');
    }
    if (sections.some((s) => !s._mk.labels.loaded() || !s._mk.exclude.loaded())) throw new Error('Masks exceed resident budget');
    return { classes: S.classes.map((c) => c.id),
      settings: { binCm: S.settings.binCm, minValid: S.settings.minValid / 100, noDataPad: S.settings.noDataPad },
      sections: serialize().cores[0].sections.map(({ name, w, h, top, bottom, units, roi, filters, detect,
        labels, exclude, stacks, objects }, i) => ({ name, w, h, top, bottom, units, roi, filters, detect,
        labels, exclude, stacks, objectCount: objects.length, eff: Array.from(effClassArray(sections[i])) })) };
  });
}

async function bins(page) {
  return page.evaluate(() => window.ichantool.S.sections.map((s) => s.bins.map((b) => [
    b.dTop, b.dBottom, b.depth, 100 * b.valid / ((b.r1 - b.r0) * (s.roi.x1 - s.roi.x0)),
    b.pct, ...b.classPct, b.BI, b.ii,
  ])));
}

async function measure(page, binCm) {
  const result = await page.evaluate(async (binCm) => {
    const { S, analyze } = window.ichantool;
    const { effClassArray, pxPerCm } = await import('./js/app.js');
    const { rowCounts, binRows } = await import('./js/core.js');
    S.settings.binCm = binCm;
    if (S.settings.noDataPad !== 0 || S.settings.minValid !== 20) throw new Error('Unexpected analysis settings');
    const timed = (operation) => { const start = performance.now(); const value = operation(); return { value, ms: performance.now() - start }; };
    for (const section of S.sections) delete section._rc;
    const cold = timed(() => { for (const section of S.sections) analyze(section); });
    const reference = JSON.stringify(S.sections.map((s) => s.bins));
    const cachedPasses = 20;
    const cached = timed(() => {
      for (let pass = 0; pass < cachedPasses; pass++) for (const section of S.sections) analyze(section);
    });
    if (JSON.stringify(S.sections.map((s) => s.bins)) !== reference) throw new Error('Cached analysis changed results');
    const sections = [];
    for (const s of S.sections) {
      const classes = timed(() => effClassArray(s));
      const rows = timed(() => rowCounts({ labels: s.labels, width: s.w, roi: s.roi, exclude: s.exclude,
        effClass: classes.value, nClasses: S.classes.length, stacks: s.stacks }));
      const aggregate = timed(() => binRows(rows.value, { top: s.top, bottom: s.bottom,
        binPx: Math.max(1, binCm * pxPerCm(s)), minValid: S.settings.minValid / 100 }));
      // Compare full integer counts as well as shared scientific output, outside timed regions.
      const previous = s._rc.counts;
      for (const key of ['valid', 'bio', 'cls']) {
        if (rows.value[key].length !== previous[key].length || rows.value[key].some((v, i) => v !== previous[key][i])) {
          throw new Error('Direct row counts differ from application cache');
        }
      }
      const shared = (b) => [b.index, b.r0, b.r1, b.dTop, b.dBottom, b.depth, b.valid, b.pct, b.classPct, b.BI, b.ii];
      if (JSON.stringify(aggregate.value.map(shared)) !== JSON.stringify(s.bins.map(shared))) throw new Error('Direct bins differ');
      sections.push({ name: s.name, roi_pixels: rows.value.nRows * rows.value.nCols, intervals: s.bins.length,
        class_mapping_ms: classes.ms, row_counts_ms: rows.ms, bin_rows_ms: aggregate.ms });
    }
    return { bin_cm: binCm, cold_ms: cold.ms, cached_ms: cached.ms / cachedPasses, cached_passes: cachedPasses,
      class_mapping_ms: sections.reduce((sum, s) => sum + s.class_mapping_ms, 0),
      row_counts_ms: sections.reduce((sum, s) => sum + s.row_counts_ms, 0),
      bin_rows_ms: sections.reduce((sum, s) => sum + s.bin_rows_ms, 0), sections };
  }, binCm);
  return { ...result, bins: await bins(page) };
}

if (process.argv[1] && resolve(process.argv[1]) === fileURLToPath(import.meta.url)) await main();

def prepare_inputs(snapshot):
    inputs = []
    for section in snapshot["sections"]:
        labels = unpack(section["labels"], section)
        if labels.size == 0 or labels.min() < 0 or labels.max() >= len(section["eff"]):
            raise ValueError("Analysis fixtures require nonoverlapping, mapped object labels")
        lookup = np.asarray(section["eff"], dtype=np.int32) + 1
        mask = lookup[labels]
        if mask.min() < 0 or mask.max() > len(snapshot["classes"]):
            raise ValueError("Effective class mapping is outside the fixture class list")
        inputs.append((mask, unpack(section["exclude"], section).astype(bool), section))
    return inputs


def analyze_inputs(inputs, snapshot, bin_cm):
    return [analyze_mask(mask, section["top"], section["bottom"], units=section["units"], bin_cm=bin_cm,
                         min_valid=snapshot["settings"]["minValid"], keep_partial=True,
                         class_names=snapshot["classes"], exclude=exclude, name=section["name"])
            for mask, exclude, section in inputs]


def result_bins(results, classes):
    columns = ["depth_top", "depth_bottom", "depth", "valid_pct", "bioturbation_pct",
               *(f"{name}_pct" for name in classes), "BI", "ii"]
    return [json.loads(result.data[columns].to_json(orient="values", double_precision=15)) for result in results]


def input_digest(inputs):
    hashed = hashlib.sha256()
    for mask, exclude, _ in inputs:
        hashed.update(mask.tobytes())
        hashed.update(exclude.tobytes())
    return hashed.hexdigest()


def profile_sites(profiler):
    sites = []
    for (filename, line, function), (primitive, calls, own, cumulative, _) in pstats.Stats(profiler).stats.items():
        path = Path(filename)
        try:
            source = path.resolve().relative_to(ROOT).as_posix()
        except ValueError:
            source = "/".join(path.parts[-3:]) if filename != "~" else "(native)"
        sites.append({"source": source, "line": line, "function": function, "primitive_calls": primitive,
                      "calls": calls, "self_ms": own * 1000, "cumulative_ms": cumulative * 1000})
    return sorted(sites, key=lambda site: site["self_ms"], reverse=True)


def python_worker(snapshot_file, profiled=False):
    snapshot = json.loads(snapshot_file.read_text(encoding="utf-8"))
    inputs = prepare_inputs(snapshot)
    before = input_digest(inputs)
    scenarios = []
    for bin_cm in BIN_CM:
        cpu_start, start = process_time(), perf_counter()
        results = analyze_inputs(inputs, snapshot, bin_cm)
        elapsed_ms, cpu_ms = (perf_counter() - start) * 1000, (process_time() - cpu_start) * 1000
        bins = result_bins(results, snapshot["classes"])
        sections = []
        for item in inputs:
            start = perf_counter()
            result = analyze_inputs([item], snapshot, bin_cm)
            sections.append({"name": item[2]["name"], "elapsed_ms": (perf_counter() - start) * 1000})
            if result_bins(result, snapshot["classes"])[0] != bins[len(sections) - 1]:
                raise ValueError("Per-section analysis changed results")
        scenario = {"bin_cm": bin_cm, "elapsed_ms": elapsed_ms, "cpu_ms": cpu_ms, "sections": sections, "bins": bins}
        if profiled:
            profiler = cProfile.Profile()
            passes = 5
            profiler.enable()
            for _ in range(passes):
                results = analyze_inputs(inputs, snapshot, bin_cm)
            profiler.disable()
            if result_bins(results, snapshot["classes"]) != bins:
                raise ValueError("Python profiling changed results")
            scenario["cpu"] = {"passes": passes, "sites": profile_sites(profiler)}
        scenarios.append(scenario)
    if input_digest(inputs) != before:
        raise ValueError("Python analysis changed the input masks")
    return {"workerPid": os.getpid(), "scenarios": scenarios, "inputs_unchanged": True}


def summarize(runs):
    import statistics

    measured = [run for run in runs if run["kind"] == "measurement"]
    summaries = []
    for index, bin_cm in enumerate(BIN_CM):
        metrics = {}
        for runtime, names in (("javascript", METRICS[:5]), ("python", METRICS[5:])):
            metrics[runtime] = {}
            for name in names:
                values = [run[runtime][index][name] for run in measured]
                metrics[runtime][name] = {"median": statistics.median(values), "min": min(values), "max": max(values)}
        summaries.append({"bin_cm": bin_cm, **metrics})
    memory = {}
    for name in ("browser_peak_rss_mib", "python_peak_rss_mib"):
        values = [run[name] for run in measured]
        memory[name] = {"median": statistics.median(values), "min": min(values), "max": max(values)}
    return {"scenarios": summaries, **memory}


def interval_differences(browser, python):
    """Keep scientific disagreements visible instead of relaxing the numerical tolerance."""
    differences = []
    for section, (js_rows, py_rows) in enumerate(zip(browser, python, strict=True)):
        if len(js_rows) != len(py_rows):
            differences.append({"section": section, "kind": "interval_count", "javascript": len(js_rows), "python": len(py_rows)})
        for row, (js, py) in enumerate(zip(js_rows, py_rows)):
            for column, (a, b) in enumerate(zip(js, py, strict=True)):
                equal = a == b if column >= len(js) - 2 or a is None or b is None else abs(a - b) <= ABS_TOL
                if not equal:
                    differences.append({"section": section, "interval": row, "column": column,
                                        "javascript": a, "python": b})
    return differences


def validate_results(browser, python, reference=None, *, diagnose_fractional=False):
    if [scenario["bin_cm"] for scenario in browser["scenarios"]] != list(BIN_CM):
        raise ValueError("Unexpected JavaScript interval settings")
    if [scenario["bin_cm"] for scenario in python["scenarios"]] != list(BIN_CM):
        raise ValueError("Unexpected Python interval settings")
    if not browser["inputs_unchanged"] or not python["inputs_unchanged"]:
        raise ValueError("Analysis did not preserve inputs")
    fingerprints, parity = [], []
    for js, py in zip(browser["scenarios"], python["scenarios"], strict=True):
        try:
            parity.append({"passed": True, **verify_bins(js["bins"], py["bins"])})
        except ValueError as error:
            if not diagnose_fractional or js["bin_cm"] != 1.37:
                raise
            parity.append({"passed": False, "first_error": str(error),
                           "differences": interval_differences(js["bins"], py["bins"]),
                           "javascript_bins": js["bins"], "python_bins": py["bins"]})
        fingerprints.append(digest(json.dumps(js["bins"], separators=(",", ":")).encode()))
    if reference is not None and fingerprints != reference:
        raise ValueError("JavaScript interval results changed between trials")
    return fingerprints, parity

    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--workspace", type=Path)

        parser.error("warmups and repetitions must be positive")

    report = {"started_utc": datetime.now(timezone.utc).isoformat(), "application": initial,

                    report["cases"].append(measure_case(case, args, url, Path(temp)))

            finally:
                server.shutdown()
                thread.join()
    if provenance() != initial:
        raise RuntimeError("Application or benchmark source changed during measurement")
    report["complete"] = True
    report["scientific_parity_passed"] = all(case["scientific_parity_passed"] for case in report["cases"])
    write_report(args.output, report)

async function measure(page) {

  await page.waitForSelector(button, { visible: true });
  await page.waitForFunction((name) => window.ichantool.S.name === name &&

    document.querySelector(selector).click();
  }, button);

  const edit = await page.evaluate((sectionCount) => {
    const { S } = window.ichantool, s = S.sections[0];
    if (S.sections.length !== sectionCount || S.sections.some((section) => !section.objects.size)) {
      throw new Error('Prepared project is missing sections or trace masks');
    }
    const id = Math.min(...s.objects.keys()), bbox = s.objects.get(id).bbox;
    if (!bbox || S.cur !== 0) throw new Error('The fixture must open on an editable first section');
    S.selected = id;
    const status = document.querySelector('#saveState');
    const start = performance.now();
    document.body.dispatchEvent(new KeyboardEvent('keydown', { key: 'Delete', bubbles: true }));
    if (s.objects.has(id) || status.dataset.state !== 'dirty') throw new Error('Delete did not dirty the project');
    return new Promise((resolve, reject) => {
      const observer = new MutationObserver(() => {
        if (status.dataset.state === 'saved' || status.dataset.state === 'error') {
          observer.disconnect();
          clearTimeout(timer);
          if (status.dataset.state === 'error') reject(new Error('Autosave failed'));
          else resolve({ id, bbox, count: s.objects.size, saveMs: performance.now() - start });
        }
      });
      const timer = setTimeout(() => {
        observer.disconnect();
        reject(new Error('Autosave did not complete'));
      }, 120_000);
      observer.observe(status, { attributes: true, attributeFilter: ['data-state'] });
    });
  }, caseId === 'u1385' ? 6 : 8);

  const analysis = await page.evaluate(() => {
    const { S, analyze } = window.ichantool;
    if (S.settings.noDataPad !== 0 || S.settings.binCm !== 2 || S.settings.minValid !== 20) {
      throw new Error('Unexpected analysis settings');
    }
    // Match the Python boundary: resident masks, then full interval computation with cold row counts.
    for (const s of S.sections) {
      if (s.labels.length !== s.w * s.h || s.exclude.length !== s.w * s.h) throw new Error('Invalid masks');
      delete s._rc;
    }
    if (S.sections.some((s) => !s._mk.labels.loaded() || !s._mk.exclude.loaded())) {
      throw new Error('Fixture exceeds the resident mask budget');
    }
    const start = performance.now();
    for (const s of S.sections) analyze(s);
    const analysisMs = performance.now() - start;
    return {
      analysisMs,
      bins: S.sections.map((s) => s.bins.map((b) => [
        b.dTop, b.dBottom, b.depth, (100 * b.valid) / ((b.r1 - b.r0) * (s.roi.x1 - s.roi.x0)),
        b.pct, ...b.classPct, b.BI, b.ii,
      ])),
    };
  });

  await page.reload({ waitUntil: 'domcontentloaded', timeout });
  await waitForProject(page, projectId);
  const snapshot = await page.evaluate(async ({ id, bbox, count, includeMasks }) => {
    const { hasId } = await import('./js/core.js');
    const { effClassArray } = await import('./js/app.js');
    const { S, serialize } = window.ichantool, s = S.sections[0];
    if (s.objects.size !== count || s.objects.has(id)) throw new Error('Deleted trace returned after reload');
    const [x0, y0, x1, y1] = bbox;
    const labels = s.labels;
    for (let y = y0; y <= y1; y++) for (let x = x0; x <= x1; x++) {
      if (hasId(labels[y * s.w + x], id, s.stacks)) throw new Error('Deleted trace pixels returned after reload');
    }
    if (!includeMasks) return null;
    const packed = serialize().cores[0].sections;
    return {
      classes: S.classes.map((c) => c.id),
      settings: { binCm: S.settings.binCm, minValid: S.settings.minValid / 100, noDataPad: S.settings.noDataPad },
      sections: packed.map(({ name, w, h, top, bottom, units, roi, filters, detect, labels, exclude, stacks, objects }, i) => ({
        name, w, h, top, bottom, units, roi, filters, detect, labels, exclude, stacks, objectCount: objects.length,
        eff: Array.from(effClassArray(S.sections[i])),
      })),
    };
  }, { ...edit, includeMasks: fixture === 'snapshot' });
  return { openMs, ...edit, ...analysis, snapshot };
}

FLAG = "SEDCORETOOL_EXPERIMENTAL_COUNTS"


def fingerprints(result):
    return [digest(json.dumps(s["bins"], separators=(",", ":")).encode()) for s in result["scenarios"]]


def summarize_pairs(pairs):
    measured = [p for p in pairs if p["kind"] == "measurement"]
    scenarios = []
    for index, size in enumerate((2, 1.37)):
        modes = {}
        for mode in ("default", "experimental"):
            values = [p[mode]["scenarios"][index]["elapsed_ms"] for p in measured]
            modes[mode] = dict(median=statistics.median(values), min=min(values), max=max(values))
        ratios = [p["experimental"]["scenarios"][index]["elapsed_ms"] /
                  p["default"]["scenarios"][index]["elapsed_ms"] for p in measured]
        ratio = modes["experimental"]["median"] / modes["default"]["median"]
        scenarios.append(dict(bin_cm=size, **modes, median_ratio=ratio, paired_ratios=ratios,
                              paired_median_ratio=statistics.median(ratios),
                              latency_passed=ratio <= .8 and statistics.median(ratios) <= .8))
    memory = {mode: statistics.median(p[mode]["peak_rss_mib"] for p in measured)
              for mode in ("default", "experimental")}
    memory_ratio = memory["experimental"] / memory["default"]
    return dict(scenarios=scenarios, median_peak_rss_mib=memory, memory_ratio=memory_ratio,
                memory_passed=memory_ratio <= 1.1,
                accepted=memory_ratio <= 1.1 and all(s["latency_passed"] for s in scenarios))


def measure_case(case, args, url, directory):
    directory.mkdir()
    seed, fixture = directory / "seed", directory / "synthetic.sedcoretool"
    print(f"Preparing {case}", file=sys.stderr, flush=True)
    if case == "synthetic":
        make_synthetic(fixture)
    subprocess.check_output(browser_args("prepare", url, seed, args.chrome, case, fixture), cwd=ROOT, timeout=TIMEOUT)
    setup = json.loads(subprocess.check_output(browser_args("measure", url, seed, args.chrome, case, "snapshot"),
                                              cwd=ROOT, timeout=TIMEOUT))
    snapshot = directory / "analysis-input.json"
    snapshot.write_text(json.dumps(setup.pop("snapshot"), separators=(",", ":")), encoding="utf-8")
    reference = json.loads((ROOT / "docs/benchmarks/M3.1.3-v0.121.json").read_text(encoding="utf-8"))
    reference = next(c for c in reference["cases"] if c["case"] == case)
    if digest(snapshot.read_bytes()) != reference["analysis_input_sha256"]:
        raise ValueError("Input fingerprint differs from corrected M0.1 fixture")
    browser = json.loads(subprocess.check_output(
        ["node", str(DRIVER), url, str(seed), args.chrome, str(snapshot), "measure"], cwd=ROOT, timeout=TIMEOUT))
    if fingerprints(browser) != reference["interval_sha256"]:
        raise ValueError("Browser results changed")
    pairs = []
    for index in range(7):
        order = ["default", "experimental"] if index % 2 == 0 else ["experimental", "default"]
        pair = dict(kind="warmup" if index == 0 else "measurement", order=order)
        for mode in order:
            result, peak, samples = run_sample([sys.executable, __file__, "--worker", str(snapshot), "--mode", mode])
            _, parity = validate_results(browser, result)
            hashes = fingerprints(result)
            if hashes != reference["python_interval_sha256"]:
                raise ValueError("Python result fingerprint changed")
            for scenario in result["scenarios"]:
                del scenario["bins"]
            pair[mode] = dict(**result, peak_rss_mib=peak, rss_samples=samples,
                              parity=parity, interval_sha256=hashes)
        pairs.append(pair)
        print(f"{case} pair {index}: {order}, "
              f"{pair['default']['scenarios'][0]['elapsed_ms']:.1f}/"
              f"{pair['experimental']['scenarios'][0]['elapsed_ms']:.1f} ms", file=sys.stderr, flush=True)
    return dict(case=case, analysis_input_sha256=digest(snapshot.read_bytes()),
                package_sha256=digest(fixture.read_bytes()) if case == "synthetic" else None,
                browser_interval_sha256=fingerprints(browser), setup_reload_verified=True,
                runtimes={k: browser[k] for k in ("chrome", "node", "puppeteer")},
                pairs=pairs, summary=summarize_pairs(pairs))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--chrome", required=True)
    parser.add_argument("--output", required=True, type=Path)
    args = parser.parse_args()
    # The driver selects each worker mode explicitly; avoid inherited experimental defaults.
    os.environ.pop(FLAG, None)
    initial = provenance()
    tool_hash = normalised_hash(Path(__file__).read_bytes())
    report = dict(started_utc=datetime.now(timezone.utc).isoformat(), application=initial,
                  driver_sha256=tool_hash, environment=describe_environment(), warmup_pairs=1,
                  measured_pairs=6, absolute_tolerance=1e-9, rss_poll_target_ms=25,
                  modes={"default": {FLAG: None}, "experimental": {FLAG: "1"}}, cases=[], complete=False)
    with tempfile.TemporaryDirectory(prefix="sedcoretool-counts-") as temp:
        with ImageServer(("127.0.0.1", 0), QuietHandler) as server:
            thread = threading.Thread(target=server.serve_forever, daemon=True)
            thread.start()
            try:
                for case in CASES:
                    report["cases"].append(measure_case(case, args, f"http://127.0.0.1:{server.server_port}/app/",
                                                        Path(temp) / case))
                    write_report(args.output, report)
            finally:
                server.shutdown()
                thread.join()
    if provenance() != initial or normalised_hash(Path(__file__).read_bytes()) != tool_hash:
        raise RuntimeError("Source changed during measurement")
    report["complete"] = True
    report["accepted"] = all(c["summary"]["accepted"] for c in report["cases"])
    write_report(args.output, report)


if __name__ == "__main__":
    if "--worker" in sys.argv:
        mode = sys.argv[sys.argv.index("--mode") + 1]
        if mode not in ("default", "experimental"):
            raise ValueError("Invalid counting mode")
        os.environ.pop(FLAG, None)
        if mode == "experimental":
            os.environ[FLAG] = "1"
        result = python_worker(Path(sys.argv[sys.argv.index("--worker") + 1]))
        result["mode"] = mode
        print(json.dumps(result))
    else:
        main()

function profileSites(profile) {
  const durations = new Map();
  (profile.samples || []).forEach((id, index) => durations.set(id, (durations.get(id) || 0) + profile.timeDeltas[index]));
  const sites = new Map();
  for (const node of profile.nodes) {
    const url = node.callFrame.url;
    const source = url.startsWith('pptr:') ? '(benchmark)' : url.split('/').slice(-2).join('/') || '(runtime)';
    const key = `${source}:${node.callFrame.lineNumber}:${node.callFrame.functionName}`;
    const site = sites.get(key) || { function: node.callFrame.functionName || '(anonymous)', source,
      line: node.callFrame.lineNumber + 1, sampled_self_ms: 0 };
    site.sampled_self_ms += (durations.get(node.id) || 0) / 1000;
    sites.set(key, site);
  }
  return [...sites.values()].filter((site) => site.sampled_self_ms > 0)
    .sort((a, b) => b.sampled_self_ms - a.sampled_self_ms).slice(0, 25);
}

"""Measure opening and interacting with the Data table on deterministic imported records."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import statistics
import subprocess
import sys
import tempfile
import threading
from datetime import datetime, timezone
from http.server import ThreadingHTTPServer
from pathlib import Path

from benchmark import ROOT, SAMPLE_SECONDS, TIMEOUT, QuietHandler, describe_environment, run_sample, source_metadata

DRIVER = ROOT / "tools" / "benchmark_data_table.mjs"
SOURCES = ("tools/benchmark_data_table.py", "tools/benchmark_data_table.mjs", "tools/data_table_fixture.mjs")
CASES = {"small": 2000, "medium": 20000, "large": 100000}
METRICS = ("open_ms", "edit_ms", "select_ms", "more_ms", "scroll_ms", "export_ms", "browser_peak_rss_mib",
           "expansion_total_ms", "last_expansion_ms", "expanded_select_ms",
           "initial_dom_rows", "expanded_dom_rows", "final_dom_rows")


class TableServer(ThreadingHTTPServer):
    request_queue_size = 32


def measure_case(case, args, url, workspace):
    def command(profile, mode):
        return ["node", str(DRIVER), url, profile, args.chrome, str(CASES[case]), mode, str(args.expansion_steps),
                "viewport" if args.viewport else "native"]

    runs, first = [], None
    for index in range(args.warmups + args.repetitions):
        with tempfile.TemporaryDirectory(prefix=f"{case}-", dir=workspace) as profile:
            result, peak, samples = run_sample(command(profile, "measure"), browser_tree=True)
        identity = {key: result[key] for key in ("fixture", "csv_sha256", "csv_rows_verified", "initial", "expanded", "displayed", "expansion_steps")}
        if first is None:
            first = identity
        elif identity != first:
            raise ValueError(f"{case} fixture or table results changed between runs")
        entry = {"kind": "warmup" if index < args.warmups else "measurement",
                 **{metric: result[metric] for metric in METRICS if metric != "browser_peak_rss_mib"},
                 "browser_peak_rss_mib": peak, "browser_rss_samples": samples,
                 "expansion_timings": result["expansion_timings"]}
        runs.append(entry)
        print(f"{case} {entry['kind']} {index + 1}: open {entry['open_ms']:.1f} ms, edit {entry['edit_ms']:.1f} ms, "
              f"more {entry['more_ms']:.1f} ms", file=sys.stderr, flush=True)
    with tempfile.TemporaryDirectory(prefix=f"{case}-cpu-", dir=workspace) as profile:
        profiled = json.loads(subprocess.check_output(command(profile, "profile"), cwd=ROOT, timeout=TIMEOUT))
    if profiled["csv_sha256"] != first["csv_sha256"]:
        raise ValueError("CPU-profile trial produced different data")
    measured = [run for run in runs if run["kind"] == "measurement"]
    return {"case": case, **first, "runs": runs,
            "runtimes": {key: result[key] for key in ("chrome", "node", "puppeteer")},
            "summary": {metric: {"median": statistics.median(run[metric] for run in measured),
                                 "min": min(run[metric] for run in measured), "max": max(run[metric] for run in measured)}
                        for metric in METRICS}, "cpu_profile": profiled["cpu"],
            "csv_cpu_profile": profiled["cpu_export"]}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--chrome", default=os.getenv("CHROME"))
    parser.add_argument("--case", choices=("all", *CASES), default="all")
    parser.add_argument("--warmups", type=int, default=1)
    parser.add_argument("--repetitions", type=int, default=5)
    parser.add_argument("--expansion-steps", type=int, default=1)
    parser.add_argument("--viewport", action="store_true", help="Enable the experimental visible-row checkbox")
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if min(args.warmups, args.repetitions, args.expansion_steps) < 1:
        parser.error("warmups, repetitions and expansion-steps must be positive")
    args.chrome = args.chrome or shutil.which("google-chrome") or shutil.which("chromium") or shutil.which("chrome")
    if not args.chrome:
        parser.error("Set --chrome to the browser executable")
    args.chrome = str(Path(args.chrome).resolve(strict=True))
    report = {"started_utc": datetime.now(timezone.utc).isoformat(), "source": source_metadata(SOURCES),
              "environment": describe_environment(), "warmups": args.warmups, "repetitions": args.repetitions,
              "rss_poll_target_ms": SAMPLE_SECONDS * 1000, "expansion_steps": args.expansion_steps,
              "viewport": args.viewport}
    with tempfile.TemporaryDirectory(prefix="sedcoretool-data-table-") as workspace:
        with TableServer(("127.0.0.1", 0), QuietHandler) as server:
            thread = threading.Thread(target=server.serve_forever, daemon=True)
            thread.start()
            try:
                url = f"http://127.0.0.1:{server.server_port}/app/index.html"
                cases = CASES if args.case == "all" else (args.case,)
                report["cases"] = [measure_case(case, args, url, workspace) for case in cases]
            finally:
                server.shutdown()
                thread.join()
    args.output.write_text(json.dumps(report, indent=2, allow_nan=False) + "\n", encoding="utf-8")


if __name__ == "__main__":
    main()

async function prepareHistoryFixture(count) {
  const { rleEncode } = await import('./js/core.js');
  const { DEFAULT_FILTERS, DEFAULT_DETECT } = await import('./js/imageproc.js');
  const width = 1024, height = 1024, columns = 96, stride = 10, tile = 6, targets = 36;
  const canvas = document.createElement('canvas'); canvas.width = width; canvas.height = height;
  const context = canvas.getContext('2d');
  context.fillStyle = '#b7a896'; context.fillRect(0, 0, width, height);
  const labels = new Int32Array(width * height), objects = [], stacks = [null];
  for (let id = 1; id <= count; id++) {
    const x = ((id - 1) % columns) * stride, y = Math.floor((id - 1) / columns) * stride;
    for (let r = y; r < y + tile; r++) labels.fill(id, r * width + x, r * width + x + tile);
    if (id <= targets) {
      labels[y * width + x] = -id;
      stacks.push([count + 1, id]);
    }
    objects.push({ id, cls: null, src: 'draw' });
  }
  objects.push({ id: count + 1, cls: null, src: 'draw' });
  const section = {
    id: 'history-section', name: 'Synthetic history section', w: width, h: height,
    ds: 1, full: { w: width, h: height }, top: 0, bottom: 100, units: 'cm',
    file: { name: 'history.png', type: 'image/png', b64: canvas.toDataURL('image/png').split(',')[1] },
    roi: { x0: 0, y0: 0, x1: width, y1: height }, filters: DEFAULT_FILTERS, detect: DEFAULT_DETECT,
    labels: rleEncode(labels), exclude: [0, labels.length], objects, nextId: count + 2, stacks,
  };
  const project = {
    app: 'SedCoreTool', kind: 'project', version: 5, id: 'benchmark-history', name: 'History benchmark',
    cores: [{ id: 'history-core', name: 'Synthetic history core', units: 'cm', sections: [section] }],
  };
  const transfer = new DataTransfer();
  transfer.items.add(new File([JSON.stringify(project)], 'history.json', { type: 'application/json' }));
  const input = document.querySelector('#impProj'); input.files = transfer.files;
  input.dispatchEvent(new Event('change', { bubbles: true }));
  return { width, height, trace_count: objects.length, tile_size_px: tile, stride_px: stride,
    columns, deletion_targets: targets, overlaps_per_target: 1, depth_cm: 100 };
}

async function waitForHistoryProject(page, count) {
  await page.waitForFunction((n) => {
    const S = window.ichantool?.S, s = S?.sections[0];
    return S?.id === 'benchmark-history' && s?.enh && s.objects.size === n
      && document.querySelector('#busy').hidden && document.querySelector('#saveState').dataset.state === 'saved';
  }, {}, count);
}

async function exerciseHistory(count, repeats) {
  const { idsAt } = await import('./js/core.js');
  const { S } = window.ichantool, s = S.sections[0];
  const undoButton = document.querySelector('#undoBtn');
  const redoButton = document.querySelector('#redoBtn');
  if (!document.querySelector('#home').hidden || !undoButton.disabled || !redoButton.disabled) throw new Error('Unexpected initial history');
  const latency = { delete: [], undo: [], redo: [] };
  const renderFrame = () => new Promise((resolve) => requestAnimationFrame(() => setTimeout(resolve, 0)));

  function remove(id) {
    S.selected = id;  // Selection uses the existing test hook. The app handles Delete normally.
    const event = new KeyboardEvent('keydown', { key: 'Delete', bubbles: true });
    const start = performance.now();
    document.body.dispatchEvent(event);
    latency.delete.push(performance.now() - start);
    if (s.objects.has(id)) throw new Error('Delete did not remove trace ' + id);
  }

  function checkTargets(deletedThrough) {
    for (let id = 1; id <= 36; id++) {
      const restored = id > deletedThrough, x0 = (id - 1) * 10;
      if (s.objects.has(id) !== restored) throw new Error('Unexpected trace ' + id);
      for (let y = 0; y < 6; y++) for (let x = 0; x < 6; x++) {
        const label = s.labels[y * s.w + x0 + x];
        if (x === 0 && y === 0 && restored) {
          const ids = idsAt(label, s.stacks);
          if (ids.length !== 2 || ids[0] !== count + 1 || ids[1] !== id) throw new Error('Overlap order changed');
        } else if (label !== (x === 0 && y === 0 ? count + 1 : restored ? id : 0)) {
          throw new Error('Unexpected pixel for trace ' + id);
        }
      }
    }
  }

  for (let id = 1; id <= 36; id++) remove(id);
  if (S.undo.length !== 30) throw new Error('Expected the 30-step history limit');
  const initial = historyShape(S.undo);
  await renderFrame();
  for (let cycle = 0; cycle < repeats; cycle++) {
    for (let id = 36; id >= 7; id--) {
      const start = performance.now();
      undoButton.click();
      latency.undo.push(performance.now() - start);
      if (!s.objects.has(id)) throw new Error('Undo did not restore trace ' + id);
    }
    if (S.undo.length || S.redo.length !== 30 || !undoButton.disabled || redoButton.disabled) throw new Error('History did not move to Redo');
    checkTargets(6);
    await renderFrame();
    for (let id = 7; id <= 36; id++) {
      const start = performance.now();
      redoButton.click();
      latency.redo.push(performance.now() - start);
      if (s.objects.has(id)) throw new Error('Redo did not delete trace ' + id);
    }
    if (S.undo.length !== 30 || S.redo.length || undoButton.disabled || !redoButton.disabled) throw new Error('History did not return to Undo');
    checkTargets(36);
    await renderFrame();
  }
  const final = historyShape(S.undo);
  if (JSON.stringify(initial) !== JSON.stringify(final)) throw new Error('History payload grew across cycles');
  for (let i = 0; i < 15; i++) undoButton.click();
  checkTargets(21);
  if (S.undo.length !== 15 || S.redo.length !== 15) throw new Error('Unexpected mixed history');
  const mixed = historyShape([...S.undo, ...S.redo]);
  return { latency, history: { initial_full: initial, final_full: final, mixed } };

  function historyShape(steps) {
    return {
      steps: steps.length, object_records: steps.reduce((n, step) => n + step.objs.size, 0),
      label_pixel_changes: steps.reduce((n, step) => n + step.lm.length / 2, 0),
      exclusion_pixel_changes: steps.reduce((n, step) => n + step.ex.length / 2, 0),
      overlap_lists: new Set(steps.flatMap((step) => step.lm.filter(Array.isArray))).size,
    };
  }
}

async function historyHeap(page, client) {
  async function used() {
    await client.send('HeapProfiler.collectGarbage');
    return (await client.send('Runtime.getHeapUsage')).usedSize;
  }
  const full = await used();
  await page.evaluate(() => {

  });
  const withoutObjects = await used();

}

function attributeHistoryCpu(profile) {
  const functions = new Set(['deleteObjects', 'beginOp', 'undo', 'redo', 'swapEditStep']);
  const parents = new Map(), nodes = new Map(), totals = {};
  for (const name of functions) totals[name] = { sample_count: 0, sampled_ms: 0 };
  for (const node of profile.nodes) {
    nodes.set(node.id, node);
    for (const child of node.children || []) parents.set(child, node.id);
  }
  for (const [i, id] of profile.samples.entries()) {
    for (let current = id; current != null; current = parents.get(current)) {
      const { functionName, url } = nodes.get(current).callFrame;
      if (url.endsWith('/app/js/app.js') && functions.has(functionName)) {
        const total = totals[functionName];
        total.sample_count++;
        total.sampled_ms += profile.timeDeltas[i] / 1000;
      }
    }
  }
  return { requested_interval_us: 100, profile_sample_count: profile.samples.length, functions: totals };
}

"""Measure SedCoreTool editing history in a disposable browser profile."""

from __future__ import annotations

import argparse
import json
import math
import os
import shutil
import statistics
import subprocess
import sys
import tempfile
import threading
from datetime import datetime, timezone
from http.server import ThreadingHTTPServer
from pathlib import Path

from benchmark import MIB, ROOT, SAMPLE_SECONDS, TIMEOUT, QuietHandler, describe_environment, run_sample, source_metadata

DRIVER = ROOT / "tools" / "benchmark_history.mjs"
HISTORY_SOURCES = ("tools/benchmark_history.py", "tools/benchmark_history.mjs")
CASES = {"sparse": 256, "dense": 8192}


class HistoryServer(ThreadingHTTPServer):
    # Leave room for Chromium's parallel requests for the app's ES modules.
    request_queue_size = 32


def summary(values):
    return {"median": statistics.median(values), "min": min(values), "max": max(values)}


def action_summary(values):
    ordered = sorted(values)
    return {"count": len(ordered), **summary(ordered), "p95": ordered[math.ceil(0.95 * len(ordered)) - 1]}


def measure_case(case, args, url, workspace):
    def command(profile, mode):
        return ["node", str(DRIVER), url, str(profile), args.chrome, str(CASES[case]), str(args.cycles), mode]

    runs, first = [], None
    for index in range(args.warmups + args.repetitions):
        with tempfile.TemporaryDirectory(prefix=f"{case}-", dir=workspace) as profile:
            result, rss, samples = run_sample(command(profile, "measure"), browser_tree=True)
        if first is None:
            first = result
        elif result["history"] != first["history"] or result["fixture"] != first["fixture"]:
            raise ValueError(f"{case} did not reproduce its fixture and history payload")
        heap = result["heap_bytes"]
        entry = {
            "kind": "warmup" if index < args.warmups else "measurement",
            "latency_ms": {action: action_summary(times) for action, times in result["latency_ms"].items()},
            "heap_bytes": heap,
            "history_heap_mib": (heap["full"] - heap["without_history"]) / MIB,
            "snapshot_heap_mib": (heap["full"] - heap["without_objects"]) / MIB,
            "other_history_heap_mib": (heap["without_objects"] - heap["without_history"]) / MIB,
            "browser_peak_rss_mib": rss, "browser_rss_samples": samples,
            "reload_verified": result["reload_verified"],
        }
        runs.append(entry)
        print(f"{case} {entry['kind']} {index + 1}: retained {entry['history_heap_mib']:.2f} MiB, "
              f"peak browser RSS {rss:.1f} MiB", file=sys.stderr)

    print(f"{case}: separate CPU profile", file=sys.stderr)
    with tempfile.TemporaryDirectory(prefix=f"{case}-profile-", dir=workspace) as profile:
        profiled = json.loads(subprocess.check_output(command(profile, "profile"), cwd=ROOT, timeout=TIMEOUT))
    if profiled["history"] != first["history"] or profiled["fixture"] != first["fixture"]:
        raise ValueError(f"{case} CPU profile ran on a different fixture or history payload")

    measured = [entry for entry in runs if entry["kind"] == "measurement"]
    metrics = ("history_heap_mib", "snapshot_heap_mib", "other_history_heap_mib", "browser_peak_rss_mib")
    timings = {
        action: {stat: summary([r["latency_ms"][action][stat] for r in measured]) for stat in ("median", "p95")}
        for action in first["latency_ms"]
    }
    return {
        "case": case, "fixture": first["fixture"], "history": first["history"],
        "runtimes": {key: first[key] for key in ("chrome", "node", "puppeteer")},
        "runs": runs, "latency_ms": timings,
        "memory_mib": {metric: summary([r[metric] for r in measured]) for metric in metrics},
        "cpu_profile": profiled["cpu"],
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--chrome", default=os.getenv("CHROME"), help="Chrome or Chromium executable (or set CHROME)")
    parser.add_argument("--case", choices=("all", *CASES), default="all")
    parser.add_argument("--cycles", type=int, default=8, help="Batches of 30 Undo and 30 Redo actions")
    parser.add_argument("--warmups", type=int, default=1)
    parser.add_argument("--repetitions", type=int, default=5)
    parser.add_argument("--output", type=Path, help="Write the JSON report here instead of stdout")
    args = parser.parse_args()
    if min(args.cycles, args.warmups, args.repetitions) < 1:
        parser.error("cycles, warmups and repetitions must be positive")
    args.chrome = args.chrome or shutil.which("google-chrome") or shutil.which("chromium") or shutil.which("chrome")
    if not args.chrome:
        parser.error("Set --chrome to a Chrome or Chromium executable")
    args.chrome = str(Path(args.chrome).resolve(strict=True))
    report = {
        "started_utc": datetime.now(timezone.utc).isoformat(), "source": source_metadata(HISTORY_SOURCES),
        "environment": describe_environment(), "cycles": args.cycles,
        "warmups": args.warmups, "repetitions": args.repetitions, "rss_poll_target_ms": SAMPLE_SECONDS * 1000,
    }
    with tempfile.TemporaryDirectory(prefix="sedcoretool-history-") as tmp:
        with HistoryServer(("127.0.0.1", 0), QuietHandler) as server:
            thread = threading.Thread(target=server.serve_forever, daemon=True)
            thread.start()
            try:
                url = f"http://127.0.0.1:{server.server_port}/app/index.html"
                cases = CASES if args.case == "all" else (args.case,)
                report["cases"] = [measure_case(case, args, url, Path(tmp)) for case in cases]
            finally:
                server.shutdown()
                thread.join()
    output = json.dumps(report, indent=2, allow_nan=False) + "\n"
    if args.output:
        args.output.write_text(output, encoding="utf-8")
    else:
        print(output, end="")


if __name__ == "__main__":
    main()

    const hash = [...new Uint8Array(await crypto.subtle.digest('SHA-256', data))].map((b) => b.toString(16).padStart(2, '0')).join('');
    return { width: s.w, height: s.h, rgba_sha256: hash, exact_pixels_verified: s.w * s.h };
  });
}
function sites(profile) {
  const weights = new Map(), result = [];
  for (let i = 0; i < (profile.samples || []).length; i++) weights.set(profile.samples[i], (weights.get(profile.samples[i]) || 0) + profile.timeDeltas[i]);
  for (const node of profile.nodes) if (weights.has(node.id)) result.push({ function: node.callFrame.functionName || '(anonymous)',
    source: node.callFrame.url.startsWith('pptr:') ? '(benchmark)' : node.callFrame.url.split('/').slice(-2).join('/') || '(runtime)',
    line: node.callFrame.lineNumber + 1, sampled_self_ms: weights.get(node.id) / 1000 });
  return result.sort((a, b) => b.sampled_self_ms - a.sampled_self_ms).slice(0, 30);
}

def normalised_hash(data):

def application_provenance():
    names = sorted(set(filter(None, git("ls-files", "-z", "--cached", "--others", "--exclude-standard", "--", "app", "src/sedcoretool").decode().split("\0"))))
    head_names = set(git("ls-tree", "-r", "--name-only", "-z", "HEAD", "--", "app", "src/sedcoretool").decode().split("\0"))
    digest = hashlib.sha256()
    mismatches = []
    for name in names:
        data = (ROOT / name).read_bytes()

        text = text or Path(name).name.startswith(("LICENSE", "OFL"))
        binary = not text
        current = hashlib.sha256(data).hexdigest() if binary else normalised_hash(data)
        head = git("show", "HEAD:" + name) if name in head_names else None
        expected = None if head is None else hashlib.sha256(head).hexdigest() if binary else normalised_hash(head)
        if current != expected:
            mismatches.append(name)
        digest.update(name.encode() + b"\0" + bytes.fromhex(current))
    delivery = re.search(r"const DELIVERY = '(V[^']+)'", (ROOT / "app/js/app.js").read_text(encoding="utf-8"))[1]
    return {"commit": git("rev-parse", "HEAD").decode().strip(), "delivery": delivery,
            "files_sha256": digest.hexdigest(), "matches_head": not mismatches, "modified_files": mismatches,
            "tool_sha256": {name: normalised_hash((ROOT / name).read_bytes()) for name in (*SOURCES, "tools/benchmark.py", "tools/benchmark_browser.mjs")}}

"""Build a single self-contained HTML file of the web app (double-click to open, works offline).

Usage:  python tools/build_standalone.py  ->  dist/SedCoreTool_V<n>.html (delivery number from app.js DELIVERY)

ES modules cannot be loaded from file:// URLs, so the modules are concatenated (in dependency
order, with import/export statements removed) into one inline <script type="module">, and the
example image is embedded as a data URI.
"""

from __future__ import annotations

import base64
import json
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
APP = ROOT / "app"
ORDER = ["core.js", "edit.js", "imageproc.js", "learn.js", "unet.js", "vision.js", "log.js", "storage.js", "zip.js", "tiff.js", "i18n.js", "state.js", "geo.js", "locmap.js", "interp.js", "plot.js", "sheet.js", "exportio.js", "tableview.js", "figview.js", "navpreview.js", "columns.js", "panels.js", "datatab.js", "figures.js", "multiproxy.js", "app.js"]

# Files the app loads with fetchAsset(path); embedded so the single file works offline.
# V0.50: no sample data in the download (no test core, example images, demos nor tutorial): only the public-domain
# coastline for the maps. The example buttons hide themselves when their files are missing.
ASSETS = ["data/land-50m.json"]
MIME = {".jpg": "image/jpeg", ".csv": "text/csv", ".json": "application/json", ".txt": "text/plain"}
TOP_DECL = re.compile(r"^(?:export\s+)?(?:async\s+)?(?:function\*?|class|const|let|var)\s+([A-Za-z_$][\w$]*)", re.M)


def strip_module(src: str) -> str:
    src = re.sub(r"^import\s[^;]*;\s*$", "", src, flags=re.M)
    return re.sub(r"^export\s+", "", src, flags=re.M)


def delivery() -> str:
    m = re.search(r"const DELIVERY = '(V\d+(?:\.\d+)*)'", (APP / "js" / "app.js").read_text(encoding="utf-8"))
    if not m:
        raise SystemExit("DELIVERY constant not found in app/js/app.js")
    return m.group(1)


def build(out: Path | None = None) -> Path:
    ver = delivery()
    out = out or ROOT / "dist" / f"SedCoreTool_{ver}.html"
    seen: dict[str, str] = {}
    parts = []
    for name in ORDER:
        code = (APP / "js" / name).read_text(encoding="utf-8")
        for ident in TOP_DECL.findall(code):
            if ident in seen:
                raise SystemExit(f"Name clash: {ident!r} defined in {seen[ident]} and {name}")
            seen[ident] = name
        parts.append(f"// ---- {name} ----\n{strip_module(code)}")
    bundle = "\n".join(parts)

    assets = {a: f"data:{MIME[Path(a).suffix]};base64," + base64.b64encode((EXAMPLES / a).read_bytes()).decode() for a in ASSETS}
    assets_js = "window.ICHANTOOL_ASSETS = " + json.dumps(assets) + ";"

    html = (APP / "index.html").read_text(encoding="utf-8").replace("<title>SedCoreTool</title>", f"<title>SedCoreTool {ver}</title>")
    # V0.40: the hosted page's CSP forbids inline scripts; the single offline file is all inline, so it gets its own
    html = re.sub(r'<meta http-equiv="Content-Security-Policy" content="[^"]*">', lambda m: m.group(0).replace("script-src 'self'", "script-src 'self' 'unsafe-inline'").replace("default-src 'self'", "default-src 'self' data: blob:"), html)
    # V0.27: the bundled fonts go inline too, so the single file looks the same offline
    fdir = APP / "vendor" / "fonts"
    fcss = re.sub(r"url\('([^']+\.woff2)'\)", lambda m: "url(data:font/woff2;base64," + base64.b64encode((fdir / m.group(1)).read_bytes()).decode() + ")", (fdir / "fonts.css").read_text(encoding="utf-8"))
    html = html.replace('<link rel="stylesheet" href="vendor/fonts/fonts.css">', "<style>" + fcss + "</style>")
    tag = '<script type="module" src="js/app.js"></script>'
    if tag not in html:
        raise SystemExit("script tag not found in app/index.html")
    html = html.replace(tag, "<script>" + assets_js + "</script>\n<script type=\"module\">\n" + bundle.replace("</script", "<\\/script") + "\n</script>")
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(html, encoding="utf-8")
    return out


if __name__ == "__main__":
    p = build()
    print(f"{p}  ({p.stat().st_size / 1e6:.2f} MB)")

/** Capture the real CSV export without downloading a file, then check every cell. Runs in the page. */
export async function captureDataTableCSV({ rowCount, edited = null, verify = true }) {
  const { parseTable } = await import('./js/log.js');
  const createURL = URL.createObjectURL;
  let csv = null;
  const preventDownload = (event) => { if (event.target.closest?.('a[download]')) event.preventDefault(); };
  document.addEventListener('click', preventDownload, true);
  URL.createObjectURL = (blob) => { if (blob.type === 'text/csv') csv = blob; return createURL(blob); };
  try {
    const start = performance.now(); document.getElementById('dCsv').click();
    if (!csv) throw new Error('Data export did not produce a CSV');
    const export_ms = performance.now() - start;
    if (!verify) return { export_ms, csv_bytes: csv.size };
    const text = await csv.text();
    const data = text.split(/\r?\n/).filter((line) => line && !line.startsWith('#')).join('\n');
    const table = parseTable(text), perCore = rowCount / 2;
    const headers = ['core', 'depth', 'age', ...Array.from({ length: 6 }, (_, column) => `value_${column}`), 'Layer 0', 'Layer 1'];
    if (table.rows.length !== rowCount || JSON.stringify(table.headers) !== JSON.stringify(headers)) throw new Error('Incomplete CSV');
    for (let index = 0; index < rowCount; index++) {
      const core = index < perCore ? 0 : 1, row = index % perCore;
      const values = Array.from({ length: 6 }, (_, column) => edited?.row === index && edited.column === column ? edited.value
        : row % 37 === 0 ? '' : ((row + 1) * (column + 3) + core + 210108) % 1009 - 504);
      const expected = [`Core ${core}`, row, row * 2, ...values,
        ...[0, 1].map((layer) => `L${layer}C${Math.floor(row / 100) % 2}`)].map(String);
      if (JSON.stringify(table.text[index]) !== JSON.stringify(expected)) throw new Error('CSV row differs: ' + index);
    }
    return { export_ms, data, csv_rows_verified: rowCount, csv_bytes: csv.size };
  } finally { URL.createObjectURL = createURL; document.removeEventListener('click', preventDownload, true); }
}

"""Generate the SYNTHETIC multi-proxy demo dataset for the IchAnTool geochemistry module.

    "DEMO-04": (39.0, 4.6, 2400), "DEMO-05": (43.4, 8.6, 1800), "DEMO-06": (40.1, 11.9, 3200),

}
SAPROPELS = [(6, 10), (79, 86), (100, 105), (122, 128)]          # low-oxygenation windows (ka)
GLACIALS = [(14, 29), (57, 71), (130, 150)]                      # detrital windows (ka)
EVENTS = ["background", "low_oxygenation", "detrital_input", "strong_bottom_current"]


def ar1(rng, n, phi=0.7, sd=0.1):
    e = rng.normal(0, sd, n)
    x = np.zeros(n)
    for i in range(1, n):
        x[i] = phi * x[i - 1] + e[i]
    return x


def make_core(rng, name, lat, lon):
    age = np.arange(0, 150.01, 0.5)
    n = len(age)
    rate = rng.uniform(5, 20)                                    # cm/ka
    depth = age * rate / 100 + rng.normal(0, 0.002, n).cumsum().clip(-0.05, 0.05)
    depth = np.maximum.accumulate(np.maximum(depth, 0))
    ev = np.zeros(n, int)
    east = np.clip((lon - 8) / 20, 0, 1)                         # sapropels stronger to the east
    for a, b in GLACIALS:
        ev[(age >= a) & (age < b)] = 2
    if lon < 10:                                                 # bottom-current pulses in the west
        for _ in range(6):
            c = rng.uniform(20, 70)
            ev[(age >= c) & (age < c + 2)] = 3
    if east > 0.15:
        for a, b in SAPROPELS:
            ev[(age >= a) & (age < b)] = 1
    lo, de, cu = ev == 1, ev == 2, ev == 3
    ti_al = 0.050 * np.exp(ar1(rng, n, sd=0.04)) * np.where(de, 1.45, 1.0)
    zr_rb = 0.80 * np.exp(ar1(rng, n, sd=0.05)) * np.where(cu, 1.7, 1.0) * np.where(de, 1.1, 1.0)
    ba_al = 0.010 * np.exp(ar1(rng, n, sd=0.05)) * np.where(lo, 1 + 2.5 * east, 1.0)
    mn_al = 0.012 * np.exp(ar1(rng, n, sd=0.06)) * np.where(lo, 0.3, 1.0)
    ca = (28 + 6 * ar1(rng, n, sd=0.15)) * np.where(de, 0.7, 1.0)
    toc = 0.45 * np.exp(ar1(rng, n, sd=0.08)) * np.where(lo, 1 + 5 * east, 1.0) * np.where(cu, 0.8, 1.0)
    rows = []
    for i in range(n):
        rows.append([name, lat, lon, round(depth[i], 4), round(age[i], 2), round(ti_al[i], 5), round(zr_rb[i], 4),
                     round(ba_al[i], 5), round(mn_al[i], 5), round(ca[i], 3), round(toc[i], 4), EVENTS[ev[i]]])
    return rows


def make_intervals(rng, name, rows):
    age = np.array([r[4] for r in rows]); depth = np.array([r[3] for r in rows])
    d = lambda a: round(float(np.interp(a, age, depth)), 4)
    edges = []
    for a, b in GLACIALS:
        edges.append((max(0.0, a + rng.uniform(-1, 1)), min(150.0, b + rng.uniform(-1, 1))))
    out, cur = [], 0.0
    for a, b in edges:
        if a > cur:
            out.append([name, d(cur), d(a), "Marga nanofosilifera", ""])
        out.append([name, d(a), d(b), "Arcilla limosa", ""])
        cur = b
    if cur < 150:
        out.append([name, d(cur), d(150), "Marga nanofosilifera", ""])
    for _ in range(int(rng.integers(0, 4))):              # turbidites only inside silty clay
        a, b = edges[int(rng.integers(0, len(edges)))]
        t0 = rng.uniform(a, max(a, b - 1.3)); t1 = t0 + rng.uniform(0.3, 1.2)
        out.append([name, d(t0), d(min(t1, b)), "", "Turbidita"])
    return out


def main():
    rng = np.random.default_rng(SEED)
    head = ["core", "lat", "lon", "depth_m", "age_ka", "Ti/Al", "Zr/Rb", "Ba/Al", "Mn/Al", "Ca_pct", "TOC_pct", "event_true"]
    (OUT / "cores").mkdir(parents=True, exist_ok=True)
    allrows, ivrows, rng_iv = [], [], np.random.default_rng(SEED + 1)
    for name, (lat, lon, _wd) in CORES.items():
        rows = make_core(rng, name, lat, lon)
        allrows += rows
        ivrows += make_intervals(rng_iv, name, rows)
        with open(OUT / "cores" / f"{name}.csv", "w", encoding="utf-8") as f:
            f.write(",".join(head[3:]) + "\n")
            for r in rows:
                f.write(",".join(map(str, r[3:])) + "\n")
    with open(OUT / "demo_long.csv", "w", encoding="utf-8") as f:
        f.write("# SYNTHETIC demo data generated by tools/make_geochem_demo.py (seed 397). Not real measurements.\n")
        f.write(",".join(head) + "\n")
        for r in allrows:
            f.write(",".join(map(str, r)) + "\n")
    with open(OUT / "catalogue.csv", "w", encoding="utf-8") as f:
        f.write("core,lat,lon,water_depth_m\n")
        for name, (lat, lon, wd) in CORES.items():
            f.write(f"{name},{lat},{lon},{wd}\n")
    with open(OUT / "intervals.csv", "w", encoding="utf-8") as f:
        f.write("core,top_m,bottom_m,Litologia,Turbidita\n")

    main()

"""Generate the SYNTHETIC training project of SedCoreTool (V19). These are NOT geological data.

Seven fictitious cores at random sea positions of the Mediterranean (seeded; positions on land are rejected with
the Natural Earth land polygons) with records driven by a simple common "climate" of the last 40 ka:
Holocene, Younger Dryas, Bølling-Allerød, Heinrich stadials HS1-HS3, Last Glacial Maximum; an organic layer
(sapropel S1 in the east, ORL in the west) 6-10 / 8-14 ka; random turbidites and two tephra layers. One core
also gets a synthetic core image (0-150 cm) with drawn trace fossils and their true ichnotaxon (truth file).

Output (examples/tutorial/): cores.csv (long table), catalogue.csv, intervals_depth.csv, intervals_age.csv,
age_models.csv, tutorial_core.jpg, tutorial_truth.json, README.md
Usage: python tools/make_tutorial.py
"""

from __future__ import annotations

import csv
import json
from pathlib import Path

import numpy as np
from PIL import Image, ImageDraw, ImageFilter

ROOT = Path(__file__).resolve().parents[1]
OUT = ROOT / "examples" / "tutorial"
SEED = 2026
# Rough sea areas (lat, lon) around which each core is placed with a random offset of up to ±0.8°.
AREAS = [("ENT-01", "Alborán", 36.0, -3.3), ("ENT-02", "Baleares", 39.3, 4.8), ("ENT-03", "Tirreno", 40.0, 12.4),
         ("ENT-04", "Jónico", 37.2, 18.3), ("ENT-05", "Adriático", 42.6, 16.0), ("ENT-06", "Egeo", 36.9, 25.2),
         ("ENT-07", "Levantino", 33.9, 30.0)]


def on_land(lat, lon, rings):
    inside = False
    for r in rings:
        xs, ys = r[0::2], r[1::2]
        if not (min(xs) <= lon <= max(xs) and min(ys) <= lat <= max(ys)):
            continue
        j = len(xs) - 1
        for i in range(len(xs)):
            if (ys[i] > lat) != (ys[j] > lat) and lon < (xs[j] - xs[i]) * (lat - ys[i]) / (ys[j] - ys[i]) + xs[i]:
                inside = not inside
            j = i
    return inside


def climate(age):
    """Cold index 0 (warm) – 1 (cold) for the last 40 ka."""
    c = np.full_like(age, 0.55)
    c[age < 11.7] = 0.05
    c[(age >= 11.7) & (age < 12.9)] = 0.75                     # Younger Dryas
    c[(age >= 12.9) & (age < 14.7)] = 0.3                      # Bølling-Allerød
    c[(age >= 14.7) & (age < 18.0)] = 0.95                     # HS1
    c[(age >= 19) & (age < 26.5)] = 0.85                       # LGM
    c[(age >= 24) & (age < 26)] = 1.0                          # HS2
    c[(age >= 29) & (age < 31)] = 0.95                         # HS3
    return c


def main():
    rng = np.random.default_rng(SEED)
    rings = json.loads((ROOT / "examples" / "data" / "land-50m.json").read_text())["rings"]
    OUT.mkdir(parents=True, exist_ok=True)
    rows, cat, ivd, iva, ages = [], [], [], [], []
    for name, area, la, lo in AREAS:
        for _ in range(200):
            lat, lon = la + rng.uniform(-0.8, 0.8), lo + rng.uniform(-0.8, 0.8)
            if not on_land(lat, lon, rings):
                break
        wd = int(rng.uniform(900, 3200))
        east = lon > 15
        rate = rng.uniform(12, 30)                                  # cm/ka
        length = int(min(900, 40 * rate))                           # cm
        # age model: 6 tie points with a wobbly rate
        tie_d = np.linspace(0, length, 6)
        tie_a = np.concatenate([[0], np.cumsum(np.diff(tie_d) / (rate * rng.uniform(0.75, 1.25, 5)))])
        for d, a in zip(tie_d, tie_a):
            ages.append((name, round(d, 1), round(a, 3)))
        depth = np.arange(0, length + 0.01, 2.0)
        age = np.interp(depth, tie_d, tie_a)
        cold = climate(age)
        a0, a1 = (6.0, 10.0) if east else (8.0, 14.0)
        org = ((age >= a0) & (age < a1)).astype(float) * (1.0 if east else 0.6)
        # turbidites (cm): 2-4 beds of 4-20 cm
        turb = np.zeros_like(depth)
        for _ in range(rng.integers(2, 5)):
            t0 = rng.uniform(20, length - 30)
            t1 = t0 + rng.uniform(4, 20)
            turb[(depth >= t0) & (depth < t1)] = 1
            ivd.append((name, round(t0, 1), round(t1, 1), "Litología", "Turbidita"))
        teph = np.zeros_like(depth)
        if area in ("Tirreno", "Egeo", "Jónico"):
            for ta in (3.6, 39.0):
                if ta < age[-1]:
                    td = float(np.interp(ta, age, depth))
                    teph[(depth >= td) & (depth < td + 3)] = 1
                    ivd.append((name, round(td, 1), round(td + 3, 1), "Litología", "Tefra"))
        n = len(depth)
        e = lambda s: rng.normal(0, s, n)  # noqa: B023 (used inside this iteration)
        rec = {
            "XRF Ti/Ca": 0.08 + 0.10 * cold + 0.25 * turb + 0.15 * teph + e(0.012),
            "XRF Ba/Ti": 0.6 + 2.2 * org - 0.2 * turb + e(0.08),
            "XRF Mn/Ti": 0.15 + 0.10 * (1 - cold) - 0.10 * org + e(0.02),
            "XRF Fe/Ca": 0.25 + 0.20 * cold + 0.15 * org + 0.3 * teph + e(0.03),
            "XRF Zr/Rb": 0.9 + 0.25 * cold + 0.8 * turb + e(0.05),
            "XRF Br/Cl": 0.05 + 0.10 * org + e(0.01),
            "SST_C": (21 if east else 18) - 7 * cold + e(0.5),
            "d18O_G.bulloides": 0.8 + 2.0 * cold - 0.6 * org + e(0.12),
            "CaCO3_pct": 45 - 15 * cold - 25 * turb - 30 * teph + e(2),
            "TOC_pct": 0.4 + (3.0 if east else 1.5) * org * rng.uniform(0.8, 1.1) + e(0.08),
            "Bentonicos_bajo_O2_pct": 10 + 60 * org + e(4),
            "Bioturbacion_pct": 70 - 50 * org + e(6),
        }
        disc = depth % 10 == 0                                       # discrete samples every 10 cm
        for i in range(n):
            r = {"core": name, "lat": round(lat, 3), "lon": round(lon, 3), "depth_cm": depth[i], "age_ka": round(age[i], 3)}
            for k, v in rec.items():
                if k in ("TOC_pct", "CaCO3_pct", "SST_C", "d18O_G.bulloides", "Bentonicos_bajo_O2_pct") and not disc[i]:
                    r[k] = ""
                else:
                    r[k] = round(float(max(v[i], 0) if k != "d18O_G.bulloides" else v[i]), 4)
            rows.append(r)
        cat.append((name, round(lat, 3), round(lon, 3), wd, f"Sondeo ficticio ({area}); datos sintéticos"))
        # described organic layer only in 4 cores (the classifier must find it in the other 3)
        if name in ("ENT-01", "ENT-03", "ENT-04", "ENT-07"):
            iva.append((name, 0.0, a0, "Capas orgánicas", "Fuera de capa orgánica"))
            iva.append((name, a0, a1, "Capas orgánicas", "Sapropel" if east else "ORL"))
            iva.append((name, a1, round(float(age[-1]), 2), "Capas orgánicas", "Fuera de capa orgánica"))
        for p0, p1, lab in [(0, 11.7, "Holoceno"), (11.7, 12.9, "Younger Dryas"), (12.9, 14.7, "Bølling-Allerød"), (14.7, 18.0, "HS1"), (19, 24, "Último Máximo Glacial")]:
            if p0 < age[-1]:
                iva.append((name, p0, min(p1, round(float(age[-1]), 2)), "Periodo climático", lab))
    keys = list(rows[0].keys())
    with open(OUT / "cores.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, keys)
        w.writeheader()
        w.writerows(rows)
    with open(OUT / "catalogue.csv", "w", newline="", encoding="utf-8") as f:
        csv.writer(f).writerows([("core", "lat", "lon", "water_depth_m", "description"), *cat])
    with open(OUT / "intervals_depth.csv", "w", newline="", encoding="utf-8") as f:
        csv.writer(f).writerows([("core", "top_cm", "bottom_cm", "layer", "value"), *ivd])
    with open(OUT / "intervals_age.csv", "w", newline="", encoding="utf-8") as f:
        csv.writer(f).writerows([("core", "top_ka", "bottom_ka", "layer", "value"), *iva])
    with open(OUT / "age_models.csv", "w", newline="", encoding="utf-8") as f:
        csv.writer(f).writerows([("core", "depth_cm", "age_ka"), *ages])
    core_image(rng)
    print("cores:", len(AREAS), "rows:", len(rows), "intervals:", len(ivd), "+", len(iva))


def core_image(rng, ppc=25, top=0, bottom=150, width_cm=8):
    """Synthetic core photo with dark trace fossils of four ichnotaxa; truth = centre and taxon of each trace."""
    H, W = int((bottom - top) * ppc), int(width_cm * ppc)
    base = rng.normal(0, 1, (H // 8 + 1, W // 8 + 1))
    base = np.array(Image.fromarray(((base - base.min()) / np.ptp(base) * 255).astype(np.uint8)).resize((W, H), Image.BICUBIC), float) / 255
    L = 150 + 25 * base + rng.normal(0, 4, (H, W))
    band = (np.arange(H)[:, None] >= 60 * ppc) & (np.arange(H)[:, None] < 78 * ppc)   # darker organic-rich band 60-78 cm
    L = np.where(band, L - 35, L)
    rgb = np.stack([L * 1.0, L * 0.88, L * 0.70], -1)
    img = Image.fromarray(np.clip(rgb, 0, 255).astype(np.uint8))
    d = ImageDraw.Draw(img)
    truth = []
    dark = lambda: tuple(int(v) for v in (rng.uniform(55, 75), rng.uniform(48, 62), rng.uniform(38, 50)))

    def free(x, y, r):
        return all((x - t["x"]) ** 2 + (y - t["y"]) ** 2 > (r + t["r"] + 4) ** 2 for t in truth)

    def place(r, y0=0, y1=None):
        for _ in range(300):
            x, y = rng.uniform(r + 4, W - r - 4), rng.uniform(y0 + r + 4, (y1 or H) - r - 4)
            if free(x, y, r):
                return x, y
        return None
    # Thalassinoides: large Y-shaped tubes (outside the organic band)
    for _ in range(9):
        p = place(30)
        if not p or band[int(p[1]), 0]:
            continue
        x, y = p
        c, wd = dark(), int(rng.uniform(9, 13))
        a = rng.uniform(0, np.pi)
        for ang in (a, a + 2.1, a + 4.2):
            d.line([(x, y), (x + 26 * np.cos(ang), y + 26 * np.sin(ang))], fill=c, width=wd)
        d.ellipse([x - wd, y - wd, x + wd, y + wd], fill=c)
        truth.append({"taxon": "Thalassinoides", "x": x, "y": y, "r": 30})
    # Zoophycos: long thin horizontal lenses with spreite
    for _ in range(14):
        p = place(32)
        if not p:
            continue
        x, y = p
        c, ln, th = dark(), rng.uniform(40, 60), rng.uniform(4, 6)
        d.ellipse([x - ln / 2, y - th / 2, x + ln / 2, y + th / 2], fill=c)
        truth.append({"taxon": "Zoophycos", "x": x, "y": y, "r": ln / 2})
    # Planolites: small horizontal ellipses
    for _ in range(40):
        p = place(10)
        if not p:
            continue
        x, y = p
        c, a, b = dark(), rng.uniform(7, 11), rng.uniform(4, 5.5)
        d.ellipse([x - a, y - b, x + a, y + b], fill=c)
        truth.append({"taxon": "Planolites", "x": x, "y": y, "r": a})
    # Chondrites: clusters of tiny dots (mostly in and around the organic band)
    for _ in range(60):
        p = place(4, 50 * ppc, 90 * ppc) if rng.random() < 0.7 else place(4)
        if not p:
            continue
        x, y = p
        c, r = dark(), rng.uniform(2, 3)
        d.ellipse([x - r, y - r, x + r, y + r], fill=c)
        truth.append({"taxon": "Chondrites", "x": x, "y": y, "r": 4})
    img = img.filter(ImageFilter.GaussianBlur(0.7))
    img.save(OUT / "tutorial_core.jpg", quality=88)
    (OUT / "tutorial_truth.json").write_text(json.dumps({"core": "ENT-03", "top_cm": top, "bottom_cm": bottom, "px_per_cm": ppc, "width": W, "height": H,
                                                          "traces": [{k: (round(v, 1) if isinstance(v, float) else v) for k, v in t.items()} for t in truth]}), encoding="utf-8")


if __name__ == "__main__":
    main()

"""V0.49: fixture for the automatic web–V2 comparison (tests/app/v2.test.mjs).

Decodes examples/data/idealized_core_0-500cm.jpg with Pillow (as the Python package does) and stores the V2.1 mask
(grey = 0.30 R + 0.56 G + 0.11 B ≤ 5, legacy/v2.1_console) run-length encoded
(`[value, run, value, run, …]`, row-major, as core.js::rleDecode reads it). With this decoder the web app's criterion
(L = 0.299 R + 0.587 G + 0.114 B < 6) selects exactly the same pixels, which the script checks.
Run again only if the image changes.
"""
import json
import sys
from pathlib import Path

import numpy as np

from sedcoretool.io import load_image, to_gray  # noqa: E402


def rle(mask: np.ndarray) -> list[int]:
    flat = mask.astype(np.int8).ravel()
    edges = np.flatnonzero(np.diff(flat)) + 1
    starts = np.concatenate(([0], edges))
    runs = np.diff(np.concatenate((starts, [flat.size])))
    out = np.empty(2 * len(starts), dtype=np.int64)
    out[0::2] = flat[starts]
    out[1::2] = runs
    return out.tolist()


raw = load_image(ROOT / "examples" / "data" / "idealized_core_0-500cm.jpg")
img = raw.astype(np.float64)
h, w = img.shape[:2]
v2 = to_gray(raw) <= 5   # same call as mask_from_threshold (pipeline.analyze_image)
lum = 0.299 * img[..., 0] + 0.587 * img[..., 1] + 0.114 * img[..., 2]
web = lum < 6
assert (v2 == web).all(), "V2 and web criteria differ on the Pillow-decoded image"
out = {"source": "examples/data/idealized_core_0-500cm.jpg", "decoder": "Pillow", "w": w, "h": h, "top": 0, "bottom": 500,
       "binPx": 150, "mask": rle(v2)}
dst = ROOT / "tests" / "app" / "fixtures" / "v2_idealized_masks.json"

"""V0.48: validate release assets and write a SHA-256 manifest before uploading a draft."""
from __future__ import annotations

import argparse
import hashlib
import json
import re
import subprocess
from pathlib import Path

from build_standalone import build, delivery

ROOT = Path(__file__).resolve().parents[1]


def prepare(folder: Path) -> None:
    folder = folder.resolve()
    folder.mkdir(parents=True, exist_ok=True)
    ver = delivery()
    names = [f"SedCoreTool-{ver}-{suffix}" for suffix in (
        "windows-setup.exe", "windows-portable.exe", "mac-universal.dmg",
        "linux-x86_64.AppImage", "linux-amd64.deb",
    )]
    missing = [name for name in names if not (folder / name).is_file()]
    if missing:
        raise ValueError("Missing installers: " + ", ".join(missing))
    html = build(folder / f"SedCoreTool_{ver}.html")
    names.append(html.name)
    allowed = set(names) | {"SHA256SUMS.txt", "RELEASE_MANIFEST.json"}
    extra = [p.name for p in folder.iterdir() if p.name not in allowed or not p.is_file()]
    if extra:
        raise ValueError("Unexpected release assets: " + ", ".join(extra))
    assets = []
    for name in sorted(names):
        p = folder / name
        if not p.stat().st_size:
            raise ValueError("Empty release asset: " + name)
        with p.open("rb") as stream:
            sha = hashlib.file_digest(stream, "sha256").hexdigest()
        assets.append({"name": name, "bytes": p.stat().st_size, "sha256": sha})
    commit = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip()
    if not re.fullmatch(r"[0-9a-f]{40}", commit):
        raise ValueError("Invalid Git commit")
    manifest = {"delivery": ver, "commit": commit, "assets": assets}
    (folder / "RELEASE_MANIFEST.json").write_text(json.dumps(manifest, indent=2) + "\n", encoding="utf-8")
    files = names + ["RELEASE_MANIFEST.json"]
    checks = []
    for name in sorted(files):
        with (folder / name).open("rb") as stream:
            checks.append(f"{hashlib.file_digest(stream, 'sha256').hexdigest()}  {name}")
    (folder / "SHA256SUMS.txt").write_text("\n".join(checks) + "\n", encoding="utf-8")
    print(f"{ver}: {len(files)} assets validated; SHA256SUMS.txt written; commit {commit}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("folder", type=Path)
    prepare(parser.parse_args().folder)

#!/usr/bin/env python3
"""V0.40: single version source. Reads DELIVERY from app/js/app.js (V0.<n>) and writes version 0.<n>.0 to
pyproject.toml, src/sedcoretool/__init__.py, CITATION.cff, desktop/package.json and the V0.<n> names in
desktop/RELEASE_NOTES.md (V0.61). `--check` only reports mismatches
(exit 1), as the test suite does. Run after changing DELIVERY."""
import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
REPO = "https://github.com/SantiCa/SedCoreTool"


def delivery() -> str:
    m = re.search(r"const DELIVERY = '(V\d+(?:\.\d+)*)'", (ROOT / "app/js/app.js").read_text(encoding="utf-8"))
    if not m:
        sys.exit("DELIVERY not found in app/js/app.js")
    parts = (re.findall(r"\d+", m.group(1)) + ["0", "0"])[:3]
    return ".".join(str(int(p)) for p in parts)


def targets(v: str):
    yield "pyproject.toml", r'(?m)^version = "[^"]*"', f'version = "{v}"'
    yield "src/sedcoretool/__init__.py", r'(?m)^__version__ = "[^"]*"', f'__version__ = "{v}"'
    yield "CITATION.cff", r"(?m)^version: .*$", f"version: {v}"


def main(check: bool) -> int:
    v, bad = delivery(), []
    for rel, pat, new in targets(v):
        p = ROOT / rel
        txt = p.read_text(encoding="utf-8")
        out = re.sub(pat, new, txt, count=1)
        if out != txt:
            bad.append(rel)
            if not check:
                p.write_text(out, encoding="utf-8")
    pj = ROOT / "desktop/package.json"
    d = json.loads(pj.read_text(encoding="utf-8"))
    want = {"version": v, "license": "GPL-3.0-or-later", "homepage": REPO}
    if any(d.get(k) != x for k, x in want.items()):
        bad.append("desktop/package.json")
        if not check:
            d.update(want)
            pj.write_text(json.dumps(d, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    lk = ROOT / "desktop/package-lock.json"
    if lk.exists():
        L = json.loads(lk.read_text(encoding="utf-8"))
        root = L.get("packages", {}).get("", {})
        if L.get("version") != v or root.get("version") != v or root.get("license") != "GPL-3.0-or-later":
            bad.append("desktop/package-lock.json")
            if not check:
                L["version"] = v
                root.update({"version": v, "license": "GPL-3.0-or-later"})
                lk.write_text(json.dumps(L, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    notes = ROOT / "desktop/RELEASE_NOTES.md"
    if notes.exists():
        tag = "V" + ".".join(v.split(".")[:2])
        txt = notes.read_text(encoding="utf-8")
        out = re.sub(r"V0\.\d+", tag, txt)
        if out != txt:
            bad.append("desktop/RELEASE_NOTES.md")
            if not check:
                notes.write_text(out, encoding="utf-8")
    if check and bad:
        print(f"version {v} not set in: {', '.join(bad)} (run tools/sync_version.py)")
        return 1
    print(f"version {v}" + (f" written to {', '.join(bad)}" if bad and not check else " consistent"))
    return 0


if __name__ == "__main__":
    sys.exit(main("--check" in sys.argv))

import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const ts = createRequire(import.meta.url)('typescript');
const root = fileURLToPath(new URL('../', import.meta.url));
const configPath = path.join(root, 'tsconfig.json');
const baselinePath = path.join(root, 'tools/typecheck-baseline.json');
const slash = (text) => text.replaceAll('\\', '/');
const relative = (file) => slash(path.relative(root, file));
const strictFiles = new Set(['app/js/core.js', 'app/js/state.js', 'app/js/imageproc.js']);
const host = { getCurrentDirectory: () => root, getCanonicalFileName: (f) => f, getNewLine: () => '\n' };

function diagnosticKey(diagnostic) {
  const { file, start, code, messageText } = diagnostic;
  if (!file || start === undefined) return null;
  const name = relative(file.fileName);
  const { line } = file.getLineAndCharacterOfPosition(start), lines = file.getLineStarts();
  const sourceLine = file.text.slice(lines[line], lines[line + 1]).trim();
  const message = slash(ts.flattenDiagnosticMessageText(messageText, '\n')).replaceAll(slash(root), './');
  const hash = createHash('sha256').update(sourceLine).update('\0').update(message).digest('hex').slice(0, 20);
  return `${name}:${code}:${hash}`;
}

const config = ts.readConfigFile(configPath, ts.sys.readFile);
const parsed = config.error ? null : ts.parseJsonConfigFileContent(config.config, ts.sys, root, {}, configPath);
const configErrors = config.error ? [config.error] : parsed.errors;
if (configErrors.length) {
  process.stderr.write(ts.formatDiagnostics(configErrors, host));
  process.exit(1);
}
const program = ts.createProgram(parsed.fileNames, parsed.options);
const diagnostics = ts.getPreEmitDiagnostics(program).filter((d) => d.category === ts.DiagnosticCategory.Error);

const baseline = JSON.parse(readFileSync(baselinePath, 'utf8'));
const remaining = new Map(Object.entries(baseline));
const newErrors = diagnostics.filter((d) => {
  const key = diagnosticKey(d), count = remaining.get(key) || 0;
  if (!count || strictFiles.has(relative(d.file.fileName))) return true;
  remaining.set(key, count - 1);
  return false;
});
if (newErrors.length) {
  process.stderr.write(ts.formatDiagnostics(newErrors, host));
  process.exitCode = 1;
} else {
  const unused = [...remaining.values()].reduce((sum, count) => sum + count, 0);
  if (unused && process.argv.includes('--prune-baseline')) {
    const active = Object.fromEntries(Object.entries(baseline)
      .map(([key, count]) => [key, count - remaining.get(key)])
      .filter(([, count]) => count > 0));
    writeFileSync(baselinePath, JSON.stringify(active, null, 2) + '\n');
  }
  console.log(`JavaScript typecheck passed (${diagnostics.length} existing diagnostics; ${unused} resolved baseline entries).`);
}

// Compare complete public-core scientific state across two application versions, excluding random section IDs.
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile } from 'node:fs/promises';
import puppeteer from 'puppeteer-core';

const [legacyURL, previewURL, executablePath = process.env.CHROME] = process.argv.slice(2);
if (!legacyURL || !previewURL || !executablePath) throw new Error('Usage: node tools/verify_navigation_equivalence.mjs LEGACY_URL PREVIEW_URL CHROME');
const browser = await puppeteer.launch({ executablePath, headless: true, args: ['--no-sandbox'], defaultViewport: { width: 1440, height: 900 } });
try {
  const legacy = await verify(legacyURL, false), preview = await verify(previewURL, true);
  assert.deepEqual(preview.hashes, legacy.hashes, 'Public-core masks, assignments, intervals or source images differ');
  console.log(JSON.stringify({ kind: 'author correctness verification, separate from performance trials',
    tool_sha256: createHash('sha256').update((await readFile(new URL(import.meta.url), 'utf8')).replaceAll('\r\n', '\n')).digest('hex'),
    excluded_field: 'random section ID only; all sections remain in source order',
    exact_scientific_hashes_equal: true, legacy, preview }, null, 2));
} finally { await browser.close(); }

async function verify(url, preview) {
  const context = await browser.createBrowserContext(), page = await context.newPage(), errors = [];
  page.setDefaultTimeout(120000); page.on('pageerror', (error) => errors.push(error.message));
  try {
    await page.goto(url, { waitUntil: 'networkidle0' });
    await page.waitForFunction(() => window.ichantool);
    await page.evaluate((enabled) => {
      const checkbox = document.getElementById('navPreview');
      if (checkbox && checkbox.checked !== enabled) checkbox.click();
      else if (enabled && !checkbox) throw new Error('Navigation preview is unavailable');
    }, preview);
    await page.$eval('#openTest', (button) => button.click());
    const ready = () => page.waitForFunction(() => window.ichantool.S.sections.length === 6 && window.ichantool.S.sections[0].enh
      && document.getElementById('busy').hidden && document.getElementById('saveState').dataset.state === 'saved');
    await ready();
    await page.$eval('#binCm', (input) => { input.value = '2'; input.dispatchEvent(new Event('change', { bubbles: true })); });
    await page.evaluate(() => window.ichantool.saveToDB()); await ready();
    const initial = await snapshot(page), sectionIDs = await page.evaluate(() => window.ichantool.S.sections.map((s) => s.id));
    await page.evaluate(() => {
      const I = window.ichantool; I.S.selected = Math.min(...I.S.sections[0].objects.keys());
      document.body.dispatchEvent(new KeyboardEvent('keydown', { key: 'Delete', bubbles: true }));
    });
    const deleted = await snapshot(page);
    assert.notEqual(deleted.science, initial.science, 'Deletion must change the scientific state');
    await page.click('#undoBtn'); assert.deepEqual(await snapshot(page), initial, 'Undo must restore every mask/result');
    await page.click('#redoBtn'); assert.deepEqual(await snapshot(page), deleted, 'Redo must restore the deletion');
    await page.evaluate(() => window.ichantool.saveToDB());
    await page.reload({ waitUntil: 'networkidle0' }); await ready();
    const reloaded = await snapshot(page); assert.deepEqual(reloaded, deleted, 'Reload must preserve the edited state');
    assert.deepEqual(errors, []);
    return { title: await page.title(), section_ids: sectionIDs, hashes: { initial, deleted, reloaded } };
  } finally { await context.close(); }
}

async function snapshot(page) {
  return page.evaluate(async () => {
    const I = window.ichantool, digest = async (bytes) => [...new Uint8Array(await crypto.subtle.digest('SHA-256', bytes))].map((b) => b.toString(16).padStart(2, '0')).join('');
    const sections = I.S.sections.map((s) => {
      I.analyze(s);
      return { w: s.w, h: s.h, ds: s.ds, top: s.top, bottom: s.bottom, units: s.units, roi: s.roi,
        labels: s._mk.labels.rle(), exclude: s._mk.exclude.rle(), stacks: s.stacks,
        objects: [...s.objects.values()].sort((a, b) => a.id - b.id).map((o) => [o.id, o.cls, o.pred, o.conf]), bins: s.bins };
    });
    const state = { settings: I.S.settings, sections }, first = I.S.sections[0];
    return { science: await digest(new TextEncoder().encode(JSON.stringify(state))),
      original: await digest(first.orig), enhanced: await digest(first.enh) };
  });
}
